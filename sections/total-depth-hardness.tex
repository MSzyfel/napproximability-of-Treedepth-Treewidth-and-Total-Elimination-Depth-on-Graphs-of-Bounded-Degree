\subsection{Total elimination depth}
\label{sec:total-depth}

The same layered graph $S_q(G)$ gives a gap for total elimination
depth, also called VPT-sum~\cite{HogemoEtAl2021}. Here every child
contributes to the objective, and each vertex of the common prefix
increases the depth of every vertex below it. We reuse the
first-branch structure and bound these contributions together.

\begin{theorem}\label{thm:total-depth-ugc}
For every fixed real $1\leq r<8/7$, approximating $\totd(G)$ within
factor $r$ on simple unweighted graphs is Unique Games-hard.
\end{theorem}

\subsubsection{The YES upper bound}

The elimination forest from Lemma~\ref{lem:ugc-completeness} has a
particularly simple depth sum. Its common prefix contains all copies
of the vertices outside an independent set. The remaining copies form
one chain per layer.

\begin{lemma}[Total depth of the cover-first forest]
\label{lem:total-depth-completeness}
For every graph $G$ on $n$ vertices and every integer $q\geq1$,
\[
 \totd(S_q(G))\leq
 \frac{qn(qn+1)-q(q-1)\alphaI(G)^2}{2}.
\]
\end{lemma}

\begin{proof}
Choose a maximum independent set of $G$, and use the forest of
Lemma~\ref{lem:ugc-completeness}. A vertex's depth counts itself and
its proper ancestors. Hence the total depth is the number of
vertices plus the number of unordered pairs that are comparable
in the ancestor order.

In this forest, the only incomparable pairs have their endpoints in
different lower chains. There are $q$ such chains, each with
$\alphaI(G)$ vertices, so there are
$\binom q2\alphaI(G)^2$ incomparable pairs. Therefore the forest has
total depth
\[
 qn+\binom{qn}{2}-\binom q2\alphaI(G)^2
 =\frac{qn(qn+1)-q(q-1)\alphaI(G)^2}{2},
\]
which gives the claimed upper bound, also when the common prefix is empty.
\end{proof}

\subsubsection{A structural lower bound for total depth}

Use the notation from Section~\ref{subsec:ugc-branch-notation}.
The prefix contributes its own depth sum and shifts every child
depth by $|P|$. Within a child, we restore the missing pairs in
$U_j\times I_j$ so that induction applies to a full layered graph.
The next two lemmas compute these two costs.

\begin{lemma}[Prefix and child total depths]
\label{lem:ugc-total-depth-accounting}
In the first-branch notation,
\[
 \totd(F)=\frac{|P|(|P|+1)}2+|P|\sum_jN_j+\sum_j\totd(F_j).
\]
\end{lemma}

\begin{proof}
The prefix vertices have depths $1,\ldots,|P|$. Each child vertex
has depth $|P|$ greater than its depth within that child. Summing
these contributions gives
\[
 \totd(F)=\sum_{i=1}^{|P|}i+|P|\sum_jN_j+\sum_j\totd(F_j)
 =\frac{|P|(|P|+1)}2+|P|\sum_jN_j+\sum_j\totd(F_j).
\]
\end{proof}

\begin{lemma}[Restoring deleted vertices: total depth]
\label{lem:restore-total-depth}
Under the hypotheses of Lemma~\ref{lem:restore-vertices}, the
forest $F'$ constructed there satisfies
\[
 \totd(F')=\totd(F)+d(N-d)+\frac{d(d+1)}2.
\]
Consequently,
$\totd(H)\leq\totd(H-X)+d(N-d)+d(d+1)/2$.
\end{lemma}

\begin{proof}
The restoration increases each of the $N-d$ old depths by $d$,
and the new vertices have depths $1,\ldots,d$. Thus
\[
 \totd(F')=\totd(F)+d(N-d)+\sum_{i=1}^d i
 =\totd(F)+d(N-d)+\frac{d(d+1)}2.
\]
Minimizing over $F$ gives the inequality.
\end{proof}

Induction gives a squared term involving each child's layer count
and vertex-cover number. Combining the children also produces
products of these quantities from different children. The following
inequality bounds the squares and products together.
Cauchy--Schwarz will then bound its right-hand side using the total
number of layers and the sum of the child vertex-cover numbers.

\begin{lemma}[Quadratic branching inequality]
\label{lem:quadratic-branching}
For a positive integer $\nu$ and nonnegative reals $W_1,\ldots,W_\nu$,
\[
 \sum_{j=1}^\nu W_j^2+14\sum_{1\leq i<j\leq \nu}W_iW_j
 \leq\left(\sum_{j=1}^\nu\sqrt{W_j}\right)^4.
\]
\end{lemma}

Its proof, by expansion into nonnegative terms, is given in
Appendix~\ref{app:total-depth-branching}.

\begin{theorem}[Structural total-depth lower bound]
\label{thm:total-depth-structural}
For every finite simple graph $G$ on $n$ vertices and every integer
$q\geq1$,
\begin{equation}\label{eq:total-depth-structural}
 \totd(S_q(G))\geq
 \frac37q^2\vc(G)^2+\frac{qn}{2}.
\end{equation}
\end{theorem}

\begin{proof}
We induct on $q$. The empty graph satisfies the bound with both
sides zero. If $q=1$, every elimination forest is a chain. More
generally, a single-chain forest has total depth
$qn(qn+1)/2\ge(3/7)q^2\vc(G)^2+qn/2$, since $\vc(G)\le n$ and
$1/2\ge3/7$. Otherwise use the first-branch notation
from Section~\ref{subsec:ugc-branch-notation}, and put
$c=\vc(G)$, $c_j=\vc(G[U_j])$, $W_j=q_jc_j$, $L_j=q_jn_j$,
$S=\sum_jW_j$, $A=q\alphaI(G)$, and $N=qn=qc+A$.
Lemmas~\ref{lem:ugc-layer-usage} and~\ref{lem:ugc-source-supports} give
$\sum_jq_j\leq q$, $\sum_jc_j\leq c$, and $n_j\leq c_j+\alphaI(G)$;
hence $S\leq qc$ and $\sum_jL_j\leq S+A$.

To use induction in child $j$, restore its $M_j=L_j-N_j$ missing
vertices above its roots. By Lemma~\ref{lem:restore-total-depth},
the added cost is
\[
 M_jN_j+\frac{M_j(M_j+1)}2
 =\frac{(N_j+M_j)(N_j+M_j+1)-N_j(N_j+1)}2
 =\frac{L_j(L_j+1)-N_j(N_j+1)}2.
\]
The restored forest represents $S_{q_j}(G[U_j])$. Since $q_j<q$,
induction bounds its total depth below by $3W_j^2/7+L_j/2$.
Subtracting the restoration cost gives
\begin{equation}\label{eq:total-depth-child-bound}
 \totd(F_j)\geq\frac37W_j^2+\frac{L_j}{2}
 -\frac{L_j(L_j+1)-N_j(N_j+1)}2
 =\frac{N_j^2+N_j-L_j^2}{2}+\frac37W_j^2.
\end{equation}
We now add the prefix and child costs. The prefix and the
triangular terms in~\eqref{eq:total-depth-child-bound} satisfy
\[
\begin{aligned}
 &\frac{|P|(|P|+1)}2+|P|\sum_jN_j+\frac12\sum_j(N_j^2+N_j)\\
 &\qquad=\frac{(|P|+\sum_jN_j)^2+|P|+\sum_jN_j}{2}
             -\sum_{i<j}N_iN_j
 =\frac{N^2+N}{2}-\sum_{i<j}N_iN_j.
\end{aligned}
\]
Thus the triangular cost of one chain on all $N$ vertices loses
one unit for each pair in different children. To bound this loss
together with the terms $L_j^2/2$, use $N_j\le L_j$ and
$\sum_jL_j\le S+A$:
\[
 \sum_{i<j}N_iN_j+\frac12\sum_jL_j^2
 \le\sum_{i<j}L_iL_j+\frac12\sum_jL_j^2
 =\frac12\Bigl(\sum_jL_j\Bigr)^2\le\frac{(S+A)^2}{2}.
\]
Lemma~\ref{lem:ugc-total-depth-accounting} therefore yields
\begin{equation}\label{eq:total-depth-combined-children}
\begin{aligned}
 \totd(F)&\geq\frac{N^2+N}{2}-\sum_{i<j}N_iN_j
             -\frac12\sum_jL_j^2+\frac37\sum_jW_j^2\\
 &\geq\frac{N^2+N}{2}-\frac{(S+A)^2}{2}+\frac37\sum_jW_j^2.
\end{aligned}
\end{equation}
The terms involving only the $W_j$ can be grouped into the
left-hand side of Lemma~\ref{lem:quadratic-branching}. Applying it,
then Cauchy--Schwarz to $\sqrt{q_j},\sqrt{c_j}$, gives
\[
\begin{aligned}
 \frac{S^2}{2}-\frac37\sum_jW_j^2
 &=\frac1{14}\left(\sum_jW_j^2+14\sum_{i<j}W_iW_j\right)
 \leq\frac1{14}\left(\sum_j\sqrt{W_j}\right)^4\\
 &\leq\frac1{14}\left(\sum_jq_j\right)^2\left(\sum_jc_j\right)^2
 \leq\frac{q^2c^2}{14}.
\end{aligned}
\]
Expand $(S+A)^2$ in~\eqref{eq:total-depth-combined-children}, apply
this estimate, and use $AS\le Aqc$. Finally, substitute $N=qc+A$:
\[
\begin{aligned}
 \totd(F)&\geq\frac{N^2+N}{2}
 -\left(\frac{S^2}{2}-\frac37\sum_jW_j^2\right)-AS-\frac{A^2}{2}
 \geq\frac{N^2+N}{2}-\frac{q^2c^2}{14}-Aqc-\frac{A^2}{2}\\
 &=\frac{q^2c^2+2Aqc+A^2+N}{2}-\frac{q^2c^2}{14}-Aqc-\frac{A^2}{2}
 =\frac37q^2c^2+\frac N2.
\end{aligned}
\]
\end{proof}

\subsubsection{The approximation gap}

\begin{proof}[Proof of Theorem~\ref{thm:total-depth-ugc}]
Fix $1\leq r<8/7$. For the source promise of Theorem~\ref{thm:kr},
Lemma~\ref{lem:total-depth-completeness} and
Theorem~\ref{thm:total-depth-structural} give the coefficients
$A^{\Sigma}_{q,\eta}=(q^2-q(q-1)(1/2-\eta)^2)/2$ and
$B^{\Sigma}_{q,\eta}=(3/7)q^2(1-\eta)^2$.
The corresponding output satisfies
\[
 \begin{array}{ll}
 \textup{YES:}&\totd(S_q(G))\leq A^{\Sigma}_{q,\eta}n^2+qn/2,\\
 \textup{NO:}&\totd(S_q(G))\geq B^{\Sigma}_{q,\eta}n^2+qn/2.
 \end{array}
\]
At $\eta=0$, the ratio of the quadratic NO and YES coefficients
is $24q/(7(3q+1))$, which approaches $8/7$ as $q$ increases.
Thus every fixed $r<8/7$ leaves a positive separation for large
enough $q$. A sufficiently small positive source error preserves
that separation, and a large enough source order makes the
quadratic terms dominate the linear terms.
To verify these choices, take a fixed integer
$q>7r/(24-21r)$. At $\eta=0$,
\[
\begin{aligned}
 A^{\Sigma}_{q,0}
 &=\frac{q^2-q(q-1)/4}{2}=\frac{q(3q+1)}8,\\
 B^{\Sigma}_{q,0}-rA^{\Sigma}_{q,0}
 &=\frac{3q^2}{7}-\frac{rq(3q+1)}8
   =\frac{q\bigl(24q-7r(3q+1)\bigr)}{56}>0.
\end{aligned}
\]
Indeed,
$24q-7r(3q+1)=(24-21r)q-7r>0$ by the choice of $q$.
The coefficients are continuous in $\eta$, so choose a fixed rational
$\eta\in(0,1/10)$ with
$B^{\Sigma}_{q,\eta}>rA^{\Sigma}_{q,\eta}$. To absorb the remaining
linear term, choose a fixed integer
$L_0\ge1$ with
$L_0>(r-1)q/[2(B^{\Sigma}_{q,\eta}-rA^{\Sigma}_{q,\eta})]$ and take
$L_0$ disjoint copies of the source graph. Since its original order is
at least one, the new order satisfies
$n\ge L_0>(r-1)q/[2(B^{\Sigma}_{q,\eta}-rA^{\Sigma}_{q,\eta})]$.
This preserves its independence fraction and gives
$(B^{\Sigma}_{q,\eta}-rA^{\Sigma}_{q,\eta})n^2>(r-1)qn/2$.
Subtracting $r$ times the YES upper bound from the NO lower bound gives
\[
 (B^{\Sigma}_{q,\eta}-rA^{\Sigma}_{q,\eta})n^2
 +\frac{qn}{2}-r\frac{qn}{2}
 =(B^{\Sigma}_{q,\eta}-rA^{\Sigma}_{q,\eta})n^2
  -\frac{(r-1)qn}{2}>0.
\]
Thus the NO lower bound exceeds $r$ times the YES upper bound.
All parameters and the number of copies are fixed, so the reduction
is polynomial.
\end{proof}

\begin{remark}[Asymptotic sharpness of the structural constants]
Let $q=2^s$ and let $G$ be the disjoint union of $q$ cliques of order
$\ell$. View $S_q(G)$ as a $q\times q$ array of $\ell$-vertex cells,
indexed by layers and source cliques. Put the two off-diagonal
quadrants on a common chain and recurse on the two diagonal quadrants,
ending with a chain in each cell. For $N=q^2\ell$, the resulting forest has
\[
 h(F)=\frac23N+\frac\ell3,\qquad
 \totd(F)=\frac37N^2+\frac N2+\frac{q\ell^2}{14}.
\]
Together with Theorems~\ref{thm:ugc-structural} and~\ref{thm:total-depth-structural},
this gives $\td(S_q(G))/N\to2/3$ and $\totd(S_q(G))/N^2\to3/7$
as $q,\ell\to\infty$, although $\alphaI(G)/|V(G)|=1/\ell\to0$.
\end{remark}
