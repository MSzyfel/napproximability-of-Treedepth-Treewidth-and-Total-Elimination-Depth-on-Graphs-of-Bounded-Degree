\subsection{Applications to treedepth}

For the parity graph $G_F$, let $W_x^a$ be the witnesses whose
local assignments set $x=a$.

\begin{lemma}[Local factors of the parity graph]\label{lem:parity-local-factors}
The edges of $G_F$ are the union of its witness-star edges and the
bicliques $T_x^0\times T_x^1$, $T_x^0\times W_x^1$, and
$T_x^1\times W_x^0$ over all variables $x$.
The star edges have maximum degree three, every vertex belongs to
at most three local factors, and their side-size ratios are at
most $12/5$. Each $W_x^a$ meets at most two source parts.
\end{lemma}

\begin{proof}
These are the truth-block and inconsistency edges in the
construction. A truth vertex belongs to two local bicliques. A
witness belongs to one per variable of its equation, hence three.
The sizes $|T_x^a|=5d_x/6$ and $|W_x^a|=2d_x$ give the ratio
$12/5$. A witness setting $x=0$ lies in $A\cup C$, and one
setting $x=1$ lies in $B\cup C$, by the placement of the special
witnesses.
\end{proof}

\begin{lemma}[Clique factors of a line graph]\label{lem:linegraph-factors}
For an unweighted host multigraph $G$, let $U_v$ be the line-graph
vertices whose host edges are incident with $v$. The nonempty sets
$U_v$ give clique factors for $\mathcal L(G)$, and each line-graph
vertex belongs to at most two factors.
\end{lemma}

\begin{proof}
Two line-graph vertices are adjacent if and only if their host edges
share an endpoint, so
$E(\mathcal L(G))=\bigcup_v\binom{U_v}{2}$.
An edge has at most two endpoints. A loop contributes one vertex
to the factor at its endpoint.
\end{proof}

\begin{corollary}[Bounded-degree treedepth hardness]
\label{cor:td-bounded-degree}
\label{thm:np-bounded-degree}
\label{thm:ugc-bounded-degree}
\label{thm:sse-bounded-degree}
For each fixed $r\ge1$, approximating treedepth within factor $r$
on simple unweighted graphs of bounded maximum degree is
NP-hard if $r<18/17$, Unique Games-hard if $r<4/3$, and
SSE-hard for every fixed $r$.
The degree bound depends only on the excluded factor. All reductions
are deterministic.
\end{corollary}

\begin{proof}
We apply Theorem~\ref{thm:td-sparsification} separately to the three
dense constructions. Denote its error parameter by $\tau>0$.
The gap transfers use Lemma~\ref{lem:factor-gap-transfer} with
exponent $a=1$.

\emph{NP-hardness.}
Use the local factors of Lemma~\ref{lem:parity-local-factors},
the three anchor cliques, and, after omitting $z_A,z_B,z_C$, the
three anchor--part bicliques. Retain the witness-star edges.
Each vertex of $G_F$ belongs to at most three local factors and one
anchor--part biclique. Each anchor vertex belongs to its clique and
one anchor--part biclique. These bicliques have equal side sizes, so
the largest side-size ratio is $12/5$, attained by the local factors.
Thus $d_{\mathrm{fac}}=4$, $d_0=3$, and $\kappa=12/5$. The occurrence
counts $d_x$ may be unbounded.

Deleting the three added vertices
changes treedepth by at most three. The resulting dense graph has
order $18m$, YES upper bound $(17/2+\eta)m+1$, and NO lower bound
$(9-\eta)m-2$. For a fixed $r<18/17$, choose positive $\eta,\tau$
so that $9-\eta-18\tau>r(17/2+\eta)$.
The sparse YES height is at most $(17/2+\eta)m+1$. The NO height is
at least $(9-\eta)m-2-18\tau m$. Thus the NO lower bound minus
$r$ times the YES upper bound is at least
\[
 \bigl(9-\eta-18\tau-r(17/2+\eta)\bigr)m-(2+r).
\]
The coefficient of $m$ is positive by the parameter choice, so a
sufficiently large fixed source repetition makes this difference
positive.

\emph{UGC hardness.}
Use the source in Theorem~\ref{thm:bounded-degree-kr-source}. In
$S_q(G)$, replace its $q$ layer cliques and retain its cross-layer
edges. Each vertex belongs to one factor, and its retained degree
is at most $(q-1)D_\eta$. There are no biclique factors, so the
theorem applies with $d_{\mathrm{fac}}=1$ and $\kappa=1$. For $r<4/3$, choose
$q,\eta$ as in Theorem~\ref{thm:ugc-hardness}. The dense YES
height is at most $A_{q,\eta}n$, the NO height is greater than
$B_{q,\eta}n$, and $B_{q,\eta}>rA_{q,\eta}$. The graph order is
$qn$. Choose $0<\tau<(B_{q,\eta}-rA_{q,\eta})/q$. The sparsification
loss is at most $\tau qn$, so the sparse NO height is greater than
\[
 B_{q,\eta}n-\tau qn
       =(B_{q,\eta}-\tau q)n>rA_{q,\eta}n.
\]
The sparse YES height is at most the dense YES bound $A_{q,\eta}n$.

\emph{SSE hardness.}
Apply Lemma~\ref{lem:sse-unweighted-gap} with its source parameter
$\varepsilon=\eta$, and let $G^*$ be the resulting host, with $N$
edges counted with multiplicity. Lemma~\ref{lem:linegraph-factors}
verifies the factor hypotheses for $\mathcal L(G^*)$ with $d_{\mathrm{fac}}=2$,
$d_0=0$, and no biclique factors.

The dense YES height is at most $4\eta N$. Put
$\beta=(c/2)\sqrt\eta\le1/4$, where $c$ is the source expansion
constant. Lemma~\ref{lem:linegraph-separators}, with $\rho=1/2$,
gives a NO balanced-separator lower bound $\beta N$, and
Lemma~\ref{lem:balanced-by-td} gives $\td(\mathcal L(G^*))\ge\beta N$.
For a fixed $r\ge1$, choose $\eta$ small enough that
$\beta>8r\eta$, and then choose $0<\tau<\beta/2$.
The sparse NO height is at least
\[
 (\beta-\tau)N>\frac{\beta N}{2}>4r\eta N.
\]
The first inequality uses $\tau<\beta/2$, and the second uses
$\beta>8r\eta$. The sparse YES height is at most $4\eta N$.

\end{proof}

The NP-hardness assertion answers the bounded-degree conjecture of
Schirrmacher, Siebertz, and
Vigny~\cite[Conjecture~1]{SchirrmacherSiebertzVigny2025}.
Weighted treedepth hardness at maximum degree ten is proved
in~\cite{DirksEtAl2025Weighted}.

\subsection{Applications to treewidth}

\begin{lemma}[Balanced separator from treewidth]
\label{lem:balanced-by-tw}
For every finite nonempty graph $H$,
$b(H)\leq\operatorname{tw}(H)+1$~\cite[Lemma~5.3]{APW}.
\end{lemma}

\begin{corollary}[Bounded-degree treewidth hardness]
\label{cor:tw-bounded-degree}
\label{thm:tw-bounded-degree}
\label{thm:tw-ugc-bounded-degree}
\label{thm:tw-sse-bounded-degree}
For each fixed $r\ge1$, approximating treewidth within factor $r$
on simple unweighted graphs of bounded maximum degree is
NP-hard if $r<12/11$, Unique Games-hard if $r<4/3$, and
SSE-hard for every fixed $r$.
The degree bound depends only on the excluded factor. All reductions
are deterministic.
\end{corollary}

\begin{proof}
We verify the hypotheses of Theorem~\ref{thm:tw-sparsification}
for each construction and denote its error parameter by $\tau$.
We use Lemma~\ref{lem:factor-gap-transfer} with exponent $a=1$.

\emph{NP-hardness.}
Use the three global cliques of $K_R(G_F)$ and, within each
replicated copy, the local factors of
Lemma~\ref{lem:parity-local-factors}. Retain the witness-star edges.
Thus $d_{\mathrm{fac}}=4$, $d_0=3$, and $\kappa=12/5$.
Each truth block lies in one global clique, while each $W_x^a$
meets at most two. Thus every biclique side partitions into at
most two sets contained in supplied clique factors, giving $s=2$.

The dense graph has order $9Rm$. By
Lemma~\ref{lem:tw-replication}, its YES width is at most
$R(11/2+\eta)m+3m-1$, and its NO width is at least
$R(6-\eta)m-1$. For $r<12/11$, choose positive $\eta,\tau$
small enough and then the fixed integer $R$ large enough that
$6-\eta-9\tau>r(11/2+\eta+3/R)$.
Sparsification lowers the NO width by at most $9\tau Rm$. Thus the
sparse NO width is at least $R(6-\eta-9\tau)m-1$, while the sparse
YES width is at most $R(11/2+\eta)m+3m-1$. Subtracting $r$ times
the YES upper bound from the NO lower bound gives
\[
 R\bigl(6-\eta-9\tau-r(11/2+\eta+3/R)\bigr)m+r-1>0.
\]
The coefficient of $m$ is positive by the parameter choice, and
$r-1\ge0$.

\emph{UGC hardness.}
For the source of Theorem~\ref{thm:bounded-degree-kr-source}, put
$Q=P_3(G)$, using the product defined in
Lemma~\ref{lem:multipartite-product}. In $K_R(Q)$, replace its three global cliques
and retain the edges within the replicated copies of $Q$.
Each vertex belongs to one clique factor and has retained degree
at most $2D_\eta$. Thus $d_{\mathrm{fac}}=1$, and the biclique hypotheses are
vacuous. The independent-set bounds in that lemma and the source
promises give $\vc(Q)\le(3/2+3\eta)n$ in the YES case and
$\vc(Q)\ge(2-2\eta)n$ in the NO case.
The dense graph has order $3Rn$, YES width at most
$R(3/2+3\eta)n+n-1$, and NO width at least $R(2-2\eta)n-1$.
For $r<4/3$, choose $\eta,\tau$ small and then $R$ large so that
$2-2\eta-3\tau>r(3/2+3\eta+1/R)$.
The sparse NO width is at least $R(2-2\eta-3\tau)n-1$, and the
sparse YES width is at most $R(3/2+3\eta)n+n-1$. Therefore
\[
 R\bigl(2-2\eta-3\tau-r(3/2+3\eta+1/R)\bigr)n+r-1>0.
\]
Again, the parameter choice makes the coefficient of $n$ positive,
and $r-1\ge0$.

\emph{SSE hardness.}
Apply Lemma~\ref{lem:sse-unweighted-gap} with source parameter
$\varepsilon=\eta$, and let $G^*$ be the resulting host, with $N$
edges counted with multiplicity. Use $\mathcal L(G^*)$, of order $N$,
with the clique factors of Lemma~\ref{lem:linegraph-factors}:
$d_{\mathrm{fac}}=2$, $d_0=0$, and no biclique factors.

The dense YES width is at most its height bound $4\eta N$.
The NO separator bound $\beta N$, where
$\beta=(c/2)\sqrt\eta\le1/4$, gives width at least $\beta N-1$
by Lemma~\ref{lem:balanced-by-tw}. Choose $\eta$ with
$\beta>8r\eta$ and then $0<\tau<\beta/2$. The sparse NO bound
is at least $(\beta-\tau)N-1$, while the sparse YES bound is at most
$4\eta N$. The coefficient difference satisfies
$\beta-\tau-4r\eta>\beta/2-4r\eta>0$. Hence
\[
 (\beta-\tau)N-1-r(4\eta N)
 =(\beta-\tau-4r\eta)N-1.
\]
Repeating every host edge a sufficiently large fixed number of times
makes this difference positive and preserves the factor hypotheses.
\end{proof}

The SSE treewidth hardness at every fixed factor on unrestricted
graphs is due to Wu, Austrin, Pitassi, and Liu~\cite[Theorem~1.1]{APW}. The bounded-degree
statement here follows by applying our treewidth sparsification
theorem to the line-graph construction.

\subsection{Applications to total depth}

\begin{corollary}[Bounded-degree total-depth hardness]
\label{cor:total-depth-bounded-degree}
\label{thm:np-total-depth-bounded-degree}
\label{thm:total-depth-ugc-bounded-degree}
\label{cor:total-depth-sse-bounded-degree}
There is an absolute constant $\epsilon_0>0$ such that approximating
total depth within factor $1+\epsilon_0$ is NP-hard on simple
unweighted graphs of bounded maximum degree. On such graphs,
total depth is also Unique Games-hard to approximate within every
fixed factor $1\le r<8/7$, and SSE-hard to approximate within every
fixed factor $r\ge1$. The degree bound depends only on the excluded
factor. All reductions are deterministic.
\end{corollary}

\begin{proof}
Apply Theorem~\ref{thm:total-depth-sparsification}, with error
parameter $\tau>0$.
The corresponding exponent in Lemma~\ref{lem:factor-gap-transfer}
is $a=2$.

\emph{NP-hardness.}
Let $\delta_0>0$ be the relative-loss hardness constant for
Independent Set on simple cubic graphs~\cite{AlimontiKann2000}.
Choose a fixed rational
$0<\epsilon_0<\min\{1/200,\delta_0/184\}$ and an integer
$\Lambda\ge64/\epsilon_0$. For a cubic source $J$ on $n\ge4$ vertices,
put $H=H^{\Sigma}_{\Lambda n}(G(J))$. The six subgraphs used in sparsification
are the three cliques on the parts of $G(J)$ and the three complete
bipartite graphs joining each part $X\in\{A,B,C\}$ to $T_X$. The
parts have sizes $2n,9n/2,9n/2$, and
$|T_A|=|T_B|=|T_C|=\Lambda n$. A vertex
of $G(J)$ belongs to its part clique and the corresponding complete
bipartite graph. Each added vertex belongs only to that graph. Since
$\Lambda\ge5$,
the largest biclique side-size ratio is $\Lambda/2$. The retained
cross-part graph has degree at most three. Thus the theorem applies
with $d_{\mathrm{fac}}=2$, $\kappa=\Lambda/2$, and $d_0=3$.

Write $N_G=|V(G(J))|=11n$ for the base graph order and
$N_H=|V(H)|=N_G+3\Lambda n=(11+3\Lambda)n$ for the output graph order. The size condition of
Lemma~\ref{lem:np-total-depth-extraction} follows from
\[
 \frac{N_G(N_G+1)}{2\Lambda n}=\frac{11n(11n+1)}{2\Lambda n}
 =\frac{121n+11}{2\Lambda}
 \leq\frac{495}{8\Lambda}n<\epsilon_0n.
\]
Here $n\ge4$ gives $11\le(11/4)n$, and
$\Lambda\ge64/\epsilon_0$ gives
$495/(8\Lambda)\le495\epsilon_0/512<\epsilon_0$.
Also, Lemma~\ref{lem:np-total-depth-cubic} and $\alphaI(J)\le n$ give
$W(G(J))=39n/2-\alphaI(J)+3\ge37n/2+3\ge n$. Hence
$\totd(H)\ge\Lambda nW(G(J))\ge\Lambda n^2$, and
$\Lambda n^2=\Lambda N_H^2/(11+3\Lambda)^2$ by the definition of $N_H$.
Choose $\tau\le\epsilon_0\Lambda/(11+3\Lambda)^2$. Then
$\tau N_H^2\le\epsilon_0\Lambda n^2\le\epsilon_0\totd(H)$.
Let $H'$ be the sparse spanning subgraph. Since every elimination
forest of $H$ is also an elimination forest of $H'$, we have
$\totd(H')\le\totd(H)$. A $(1+\epsilon_0)$-approximate elimination
forest $F$ of $H'$ completes to an elimination forest $\widehat F$ of $H$, and
\[
\begin{aligned}
 \totd(\widehat F)
 &\leq\totd(F)+\tau N_H^2
 \leq(1+\epsilon_0)\totd(H')+\tau N_H^2\\
 &\leq(1+\epsilon_0)\totd(H)+\epsilon_0\totd(H)
 =(1+2\epsilon_0)\totd(H).
\end{aligned}
\]
Lemma~\ref{lem:np-total-depth-extraction} applies with error
$2\epsilon_0$ and yields an independent set of $J$ of size
at least $(1-184\epsilon_0)\alphaI(J)>(1-\delta_0)\alphaI(J)$,
contradicting the cubic source hardness.

\emph{UGC hardness.}
Use Theorem~\ref{thm:bounded-degree-kr-source} and the graph
$S_q(G)$. Its layer cliques are the only factors. Each vertex
belongs to one, and its retained cross-layer degree is at most
$(q-1)D_\eta$. Hence $d_{\mathrm{fac}}=1$, with no biclique condition. For
$r<8/7$, choose $q,\eta$ as in
Theorem~\ref{thm:total-depth-ugc}. Its YES bound is
$\totd(S_q(G))\le A^\Sigma_{q,\eta}n^2+qn/2$, and its NO bound is
$\totd(S_q(G))\ge B^\Sigma_{q,\eta}n^2+qn/2$, where
$B^\Sigma_{q,\eta}>rA^\Sigma_{q,\eta}$. The graph order
is $qn$. Choose
$0<\tau<(B^\Sigma_{q,\eta}-rA^\Sigma_{q,\eta})/q^2$.
After sparsification, the NO lower bound is at least
$B^\Sigma_{q,\eta}n^2+qn/2-\tau q^2n^2$, while the YES upper
bound is at most $A^\Sigma_{q,\eta}n^2+qn/2$. Subtracting $r$ times
the YES bound from the NO bound gives
\[
 \bigl(B^\Sigma_{q,\eta}-rA^\Sigma_{q,\eta}-\tau q^2\bigr)n^2
 -(r-1)qn/2.
\]
The quadratic coefficient is positive by the choice of $\tau$.
Choose the number of disjoint source copies so that, for the resulting
source order $n$,
\[
 \bigl(B^\Sigma_{q,\eta}-rA^\Sigma_{q,\eta}-\tau q^2\bigr)n^2
 >(r-1)qn/2.
\]
This makes the NO bound exceed $r$ times the YES bound. Disjoint
copies preserve the source densities and maximum degree.

\emph{SSE hardness.}
Apply Lemma~\ref{lem:sse-unweighted-gap} with source parameter
$\varepsilon=\eta$, and let $G^*$ be the resulting host, with $N$
edges counted with multiplicity. Use $\mathcal L(G^*)$, of order $N$,
with the clique factors of Lemma~\ref{lem:linegraph-factors}:
$d_{\mathrm{fac}}=2$, $d_0=0$, and no biclique factors. The dense YES height is at most
$4\eta N$, so its total depth is at most $4\eta N^2$.
For $\beta=(c/2)\sqrt\eta\le1/4$,
Lemma~\ref{lem:linegraph-separators} at $\rho=1/2$ gives a NO
balanced-separator lower bound $\beta N$. By
Lemma~\ref{lem:balanced-by-td}, this implies
$\totd(\mathcal L(G^*))\ge\beta N^2/2$.
Given a fixed $r\ge1$, choose $\eta$ so that $\beta>16r\eta$,
and then choose $0<\tau<\beta/4$. The sparse YES bound remains
$4\eta N^2$, and its NO total depth is at least
$(\beta/2-\tau)N^2>\beta N^2/4>4r\eta N^2$.
The first strict inequality uses $\tau<\beta/4$, and the second uses
$\beta>16r\eta$.
Therefore the sparse NO bound exceeds $r$ times the YES bound.

\end{proof}

\subsection{Regular graphs and subdivisions}

\begin{proposition}\label{prop:regularization}
For every graph $G$ of order $n$ and fixed odd $d\geq3$ with
$d>\Delta(G)$, one can construct in polynomial time a simple
$d$-regular graph $R$ containing $G$ as an induced subgraph, with
$(d+2)n\leq |V(R)|\leq(d+3)n$ and
\[
\begin{aligned}
 \td(G)&\leq\td(R)\leq\td(G)+d+2,\\
 \operatorname{tw}(G)&\leq\operatorname{tw}(R)
 \leq\max\{\operatorname{tw}(G),d+2\},\\
 (d+2)\totd(G)&\leq\totd(R)
 \leq(d+3)\totd(G)+\tfrac12(d+2)(d+3)n.
\end{aligned}
\]
\end{proposition}

\begin{proof}
For each $v$, set $t=d-\deg_G(v)$ and attach a fresh gadget $Q_t$
by joining $v$ to its $t$ ports. Since $d>\Delta(G)$, $t\geq1$.
If $t$ is even, take $K_{d+1}$ and remove a perfect matching on the
ports. If $t\geq3$ is odd, take $K_{d+2}$ and remove a cycle on the
ports and a perfect matching on the other vertices; the number
$d+2-t$ of other vertices is even. For $t=1$, take $K_{d+2}$,
remove a two-edge path centered at the port, and remove a perfect
matching on the remaining vertices. In each case every port has
degree $d-1$ inside $Q_t$ and gains one edge to $v$, so its final
degree is $d$; all other gadget vertices already have degree $d$.
Each $Q_t$ has minimum
degree $d-1$ and at most $d+2$ vertices. Every connected component
therefore has at least $d$ vertices, so $Q_t$ is connected because
$2d>d+2$ for $d\geq3$. The resulting graph $R$ is simple.

Appending each gadget as a chain below $v$ proves the upper bounds
for height and total depth. Adding a bag $\{v\}\cup V(Q_t)$ to a
tree decomposition proves the treewidth bound.

For the total-depth lower bound, partition $R$ into connected sets
$X_v=\{v\}\cup V(Q_t)$. In any elimination forest, the shallowest
vertex $r_v$ of $X_v$ is an ancestor of every vertex in $X_v$.
To see this, take a path in $X_v$ starting at $r_v$. If a next vertex
is a descendant of the current one, it remains below $r_v$. If it is
an ancestor of the current one, it lies on the path from the root to
the current vertex and cannot be above $r_v$, by the choice of $r_v$.
Thus every vertex of $X_v$ is below $r_v$. For every source edge $uv$,
the endpoints $u,v$ are comparable in the forest. Since $r_u$ and
$r_v$ are ancestors of $u$ and $v$, they are both ancestors of the
deeper endpoint and are therefore comparable. Restrict the forest to
the representatives $r_v$; the result $\overline F$ is an elimination
forest of $G$. Every vertex of $X_v$ has depth in the original forest
at least $\operatorname{depth}_{\overline F}(r_v)$, so
\[
 \totd(F)\geq\sum_v|X_v|\operatorname{depth}_{\overline F}(r_v)
 \geq(d+2)\totd(\overline F)\geq(d+2)\totd(G).
\]
This argument also gives a polynomial-time projection of forests.
\end{proof}

\begin{corollary}\label{cor:regular-hardness}
The hardness ranges in Corollaries~\ref{cor:td-bounded-degree},
\ref{cor:tw-bounded-degree}, and~\ref{cor:total-depth-bounded-degree}
also hold on $d$-regular graphs for a suitable fixed odd degree $d$
depending only on the excluded factor. The NP-hardness constant for
total depth may decrease.
\end{corollary}

\begin{proof}
Fix a target factor $r$ in one of the stated hardness ranges, and
choose a larger factor $r'$ that is still in the same range. For the
NP total-depth result, take the target factor below $1+\epsilon_0$;
the preceding result then also excludes approximation within any
larger factor $r'<1+\epsilon_0$.
Choose an odd $d$ larger than the maximum degree of the bounded-degree
instances and large enough that
\[
 r\frac{d+3}{d+2}<r'.
\]
Such a $d$ exists because $(d+3)/(d+2)$ tends to $1$.

Write $Y$ and $Z$ for the YES upper bound and NO lower bound on a
source instance, so $Z>r'Y$. The reductions have unbounded scaling
parameters. For treedepth, the YES bound grows linearly with the
source size, so we can make $r(Y+d+2)<Z$; Proposition~\ref{prop:regularization}
then preserves the factor-$r$ gap. For treewidth, enlarge the source
until its YES width is at least $d+2$. The proposition then leaves the
YES upper bound unchanged and cannot decrease the NO lower bound.

For total depth, let $n$ be the source order and write
$C_d=(d+2)(d+3)/2$. The proposition gives regularized YES and NO
bounds of $(d+3)Y+C_dn$ and $(d+2)Z$, respectively. The difference
between the NO bound and $r$ times the YES bound is greater than
\[
 \bigl((d+2)r'-r(d+3)\bigr)Y-rC_dn.
\]
Its quadratic coefficient is positive by the choice of $d$, and the
total-depth YES bounds in these reductions grow quadratically with
$n$. Increasing their scaling parameters therefore makes this
difference positive. The resulting degree $d$ depends only on the
target factor.
\end{proof}

\begin{corollary}\label{cor:subdivision-hardness}
For every fixed $g\geq3$, the treedepth and treewidth hardness ranges above
hold on bounded-degree graphs that are bipartite, $2$-degenerate, and
have girth at least $g$. Here $2$-degenerate means that every
nonempty subgraph has a vertex of degree at most two, and girth is the
length of a shortest cycle (infinite for an acyclic graph). On the same class, total depth is
SSE-hard to approximate within every fixed factor.
\end{corollary}

\begin{proof}
Replace every edge of $G$ by a path of fixed even length
$\ell\geq\max\{2,g/3\}$, obtaining $G_\ell$. All original vertices
lie in one side of a bipartition, since every replacement path has
even length. Each internal vertex has degree two, and deleting all
internal vertices leaves the original vertices isolated, so
$G_\ell$ is $2$-degenerate. Every cycle uses at least three
replacement paths and therefore has length at least $3\ell\geq g$.
Contracting each replacement path shows $\operatorname{tw}(G)
\leq\operatorname{tw}(G_\ell)$. Conversely, for each path
$u=x_0,x_1,\ldots,x_\ell=v$, attach the bags
$\{u,v,x_1\},\{v,x_1,x_2\},\ldots,
\{v,x_{\ell-2},x_{\ell-1}\}$ along a path to an old bag containing
$u,v$. These bags cover the replacement path and preserve the
connectedness condition for each vertex, so
$\operatorname{tw}(G_\ell)\leq\max\{\operatorname{tw}(G),2\}$.
For graphs with an edge, $\operatorname{tw}(G)\geq1$, while an
edgeless graph remains edgeless. Thus subdivision preserves
treewidth. A path on $s$ vertices has an
elimination tree of height at most $\lceil\log_2(s+1)\rceil$: choose
a middle vertex as root and recurse on the two remaining subpaths.
Thus the $\ell-1$ internal vertices on each replaced edge have such
a tree of height $h=\lceil\log_2\ell\rceil$. Attach this tree below
the deeper of the original edge's endpoints in an elimination forest
of $G$. The original endpoints are comparable, so the edge to the
shallower endpoint is represented as well. Together with
minor-monotonicity, this gives
\[
 \td(G)\leq\td(G_\ell)\leq\td(G)+h,\qquad h=\lceil\log_2\ell\rceil.
\]
Thus the linear gaps persist. For total depth, start with a
$d$-regular instance from Corollary~\ref{cor:regular-hardness}.
Since $d$ is odd, add a perfect matching to make every degree even,
using auxiliary matching edges that may be parallel to original edges.
Orient the augmented multigraph along an Euler tour in each connected
component, then delete the matching edges. Each vertex has outdegree
$(d+1)/2$ before deletion, so its outdegree in the original graph is
at least $(d-1)/2$.

Assign the internal vertices of each replacement path to the tail
of its oriented edge. For each vertex $v$, let $X_v$ consist of $v$
and the internal vertices assigned to it. The sets $X_v$ partition
$V(G_\ell)$; each is connected and has size at least
$s=1+(\ell-1)(d-1)/2$. In an elimination forest of $G_\ell$, let
$r_v$ be the shallowest vertex of $X_v$. Since $X_v$ is connected
and the endpoints of each graph edge are comparable, $r_v$ is an
ancestor of every vertex in $X_v$. Each original edge $uv$ has a
replacement path whose last edge joins $X_u$ to $X_v$, so $r_u$ and
$r_v$ are comparable. Therefore the representatives $r_v$ form an
elimination forest of $G$. Every vertex in $X_v$ has depth at least
$\operatorname{depth}_{F}(r_v)$, giving the lower bound below.

For the upper bound, take an optimal elimination forest of $G$ and,
for each original edge, attach a midpoint-recursive elimination tree
of its internal path below the deeper endpoint. Each of the
$\ell-1$ new vertices on that path has depth at most the depth of
that endpoint plus $h$. Each vertex is the deeper endpoint of at
most $d$ edges, so summing the original and new depths gives
\[
\begin{aligned}
 s\totd(G)&\leq\totd(G_\ell)\\
 &\leq\totd(G)+(\ell-1)\sum_{uv\in E(G)}
       \operatorname{depth}_{F}(\text{deeper endpoint of }uv)
       +h(\ell-1)|E(G)|\\
 &\leq[1+d(\ell-1)]\totd(G)+h(\ell-1)|E(G)|.
\end{aligned}
\]
Here $1+d(\ell-1)\leq3s$, since
$3s-[1+d(\ell-1)]=2+(\ell-1)(d-3)/2\geq0$. The last term is
$O(n)$ because $d$ and $\ell$ are fixed. Choose the source SSE gap
with NO bound $Z>3rY$, where $Y$ is its YES upper bound. The
subdivided instance then has YES bound at most $3sY+O(n)$ and NO
bound at least $sZ$. The explicit SSE gaps above have a positive
quadratic margin between $Z$ and $3rY$, so scaling the source makes
$sZ-r(3sY+O(n))>0$. Thus the factor-$r$ gap persists.
\end{proof}

The total-depth conclusions of both corollaries also hold for average
depth, by division by the graph order, and for trivially perfect
completion, by~\eqref{eq:tp-adjusted-cost}.
