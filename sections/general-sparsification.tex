\subsection{General sparsification theorems}
\label{subsec:general-sparsification}

The comparison lemmas reduce sparsification to choosing thresholds
whose deletion bounds have a small sum. We take each threshold
proportional to its factor's size. To bound the sum by a multiple of
$N=|V(H)|$, we limit how many factors contain each vertex. Together
with a bound on the degree of the retained edges, this also bounds
the degree after replacement.

Use the edge decomposition in~\eqref{eq:factor-edge-union} and the
replacement graph $J$ from Section~\ref{sec:bounded-degree-hardness}.
Fix constants $d_0\ge0$,
$d_1\ge1$, and $\kappa\ge1$, and assume:
\begin{enumerate}
\item\label{hyp:retained-degree} the retained graph $(V(H),E_0)$ has maximum degree at most $d_0$.
\item\label{hyp:factor-incidence} each vertex belongs to at most $d_1$ factor supports $U_f$.
\item\label{hyp:biclique-ratio} every biclique factor has nonempty sides and satisfies
$\max\{|A_f|,|B_f|\}/\min\{|A_f|,|B_f|\}\le\kappa$.
\end{enumerate}
Counting incidences between vertices and factor supports gives
\begin{equation}\label{eq:general-factor-incidence}
 \sum_{f\in\mathcal C\cup\mathcal B}|U_f|\le d_1N.
\end{equation}

\begin{lemma}[Common sparse factor replacements]
\label{lem:factor-replacement}
Let $H$ have the edge description~\eqref{eq:factor-edge-union},
and let fixed constants $d_0\ge0$, $d_1\ge1$, $\kappa\ge1$ satisfy
conditions~\ref{hyp:retained-degree}--\ref{hyp:biclique-ratio}.
Fix $\zeta>0$ and set
$t_f=\lceil\zeta|U_f|\rceil$ for every factor. There is a
deterministic polynomial-time construction of a spanning subgraph
$J\subseteq H$ with $E_0\subseteq E(J)$ that replaces each factor by a graph
with no $(t_f,t_f)$ hole, and has maximum degree at most
$d_0+d_1d_{\zeta,\kappa}$, where $d_{\zeta,\kappa}$
is the constant in Lemma~\ref{lem:factor-expanders}.
Moreover, $\sum_f(t_f-1)<\zeta d_1N$ whenever at least one factor
is present.
\end{lemma}

\begin{proof}
Apply Lemma~\ref{lem:factor-expanders} to each clique support and
each pair of biclique sides, using $\zeta,\kappa$. Its thresholds
are $t_f=\lceil\zeta|U_f|\rceil$ in both cases. Thus each
replacement satisfies the required no-hole condition and has
maximum degree at most $d_{\zeta,\kappa}$.
Take $E(J)$ to be $E_0$ together with the replacement edges.
For every $v\in V(H)$, the degree in the union satisfies
\[
 \deg_J(v)\le\deg_{(V(H),E_0)}(v)
 +|\{f\in\mathcal C\cup\mathcal B:v\in U_f\}|d_{\zeta,\kappa}
 \le d_0+d_1d_{\zeta,\kappa}.
\]
Finally, $t_f-1<\zeta|U_f|$ for each factor. Summing and using
\eqref{eq:general-factor-incidence} proves the last assertion.
Each factor has nonempty support, so there are at most
$d_1N$ factors. The construction is therefore
polynomial time for the fixed parameters.
\end{proof}

\begin{lemma}[Preservation of a dense hardness gap]
\label{lem:factor-gap-transfer}
Let $\varphi$ be a graph minimization parameter and fix $a>0$ and $r\ge1$.
Suppose that, for every fixed $\tau>0$, a deterministic
polynomial-time construction maps each $N$-vertex graph $H$ in
a given family to a spanning subgraph $J$ of bounded maximum degree
with $\varphi(H)-\tau N^a\le\varphi(J)\le\varphi(H)$.
The degree bound may depend on $\tau$ and the family's fixed
parameters. Suppose a hardness reduction to this family has, for
fixed $B>rA\ge0$, YES bound $\varphi(H)\le AN^a+e(N)$ and
NO bound $\varphi(H)\ge BN^a-e(N)$, where $e(N)\ge0$ and
$e(N)/N^a\to0$ as $N\to\infty$.
For every fixed $N_0$, assume the source promise remains hard when
restricted to instances mapped to graphs of order $N\ge N_0$.
Then factor-$r$ hardness holds at bounded maximum degree under the
same assumption and reduction type.

If, in addition, every feasible solution for $J$ can be completed
to one for $H$ at an additive cost of at most $\tau N^a$, and
$\varphi(H)\ge c_{\min}N^a$ for a fixed $c_{\min}>0$, then an $r$-approximate
solution for $J$ completes to a solution for $H$ within factor
$r+\tau/c_{\min}$.
\end{lemma}

\begin{proof}
Choose fixed $0<\tau<B-rA$. The sparse YES upper bound is
$AN^a+e(N)$, and the sparse NO lower bound is
$(B-\tau)N^a-e(N)$. Their difference after multiplying the YES
bound by $r$ is $(B-\tau-rA)N^a-(r+1)e(N)>0$ for every
sufficiently large $N$, since $e(N)/N^a\to0$.
For the solution statement, the assumption gives
$\varphi(J)\le\varphi(H)$. Thus completion gives cost at most
$r\varphi(J)+\tau N^a\le r\varphi(H)+\tau N^a\le(r+\tau/c_{\min})\varphi(H)$.
\end{proof}

\begin{theorem}[Treedepth sparsification]
\label{thm:td-sparsification}
Let $H$ have the edge description~\eqref{eq:factor-edge-union},
and let fixed constants $d_0\ge0$, $d_1\ge1$, $\kappa\ge1$ satisfy
conditions~\ref{hyp:retained-degree}--\ref{hyp:biclique-ratio}.
For every fixed $\varepsilon>0$
one can construct deterministically in polynomial time a spanning
subgraph $J\subseteq H$ whose maximum degree depends only on
$d_0,d_1,\kappa,\varepsilon$ and such that
$\td(H)-\varepsilon N\le\td(J)\le\td(H)$.
Every elimination forest of $J$ can be transformed into one of
$H$ whose height increases by at most $\varepsilon N$.
\end{theorem}

\begin{proof}
Apply Lemma~\ref{lem:factor-replacement} with
$\zeta=\varepsilon/(4d_1)$. Its degree bound depends only on
the stated constants. The error bound in
Lemma~\ref{lem:factor-sparsification} satisfies
\begin{equation}\label{eq:common-forest-exception-budget}
 \begin{aligned}
 \Delta_{\mathrm{td}}
 &=2\sum_{f\in\mathcal C}(t_f-1)+4\sum_{f\in\mathcal B}(t_f-1)\\
 &\le4\sum_{f\in\mathcal C\cup\mathcal B}(t_f-1)
 \le4\zeta d_1N=\varepsilon N.
 \end{aligned}
\end{equation}
Lemma~\ref{lem:factor-sparsification} now gives both conclusions.
\end{proof}

\begin{theorem}[Treewidth sparsification]
\label{thm:tw-sparsification}
Let $H$ have the edge description~\eqref{eq:factor-edge-union},
and let fixed constants $d_0\ge0$, $d_1\ge1$, $\kappa\ge1$ satisfy
conditions~\ref{hyp:retained-degree}--\ref{hyp:biclique-ratio}.
Suppose additionally that each side
of every biclique can be partitioned into at most $s$ sets, each
contained in a supplied clique factor, where $s\ge1$ is fixed.
For every fixed $\varepsilon>0$, one can construct deterministically
in polynomial time a spanning subgraph $J\subseteq H$ whose maximum
degree depends only on $d_0,d_1,\kappa,s,\varepsilon$ and such that
$\operatorname{tw}(H)-\varepsilon N
 \le\operatorname{tw}(J)\le\operatorname{tw}(H)$.
Every tree decomposition of $J$ can be transformed into one of
$H$ whose width increases by at most $\varepsilon N$.
\end{theorem}

\begin{proof}
Apply Lemma~\ref{lem:factor-replacement} with
$\zeta=\varepsilon/(d_1\max\{3,2s^2\})$. This constructs $J$ by replacing
each original factor once.

For the comparison argument, partition each biclique factor
$A_f\times B_f$ into the at most $s^2$ bicliques between pairs
of its supplied pieces. Their union is the original biclique.
Each side of every resulting biclique is contained in a supplied
clique factor, as required by
Lemma~\ref{lem:tw-factor-sparsification}. Use the original
threshold $t_f$ for every piece pair. The restriction of the sparse
replacement to that pair has no $(t_f,t_f)$ hole: any two
$t_f$-subsets of the pieces are also subsets of the original sides.
If a piece has fewer than $t_f$ vertices, this condition holds
vacuously.

Apply Lemma~\ref{lem:tw-factor-sparsification} to this factor
description of the same graphs $H,J$. There is one clique term
for each original clique, and at most $s^2$ biclique terms, each
with threshold $t_f$, for each original biclique. Bounding both
coefficients by $\max\{3,2s^2\}$ and using
$\sum_f(t_f-1)\le\zeta d_1N$ bounds the error by
\[
 3\sum_{f\in\mathcal C}(t_f-1)
       +2s^2\sum_{f\in\mathcal B}(t_f-1)
 \le\max\{3,2s^2\}\zeta d_1N=\varepsilon N.
\]
Consequently, every tree decomposition of $J$ completes to one
of $H$ with width increase at most $\varepsilon N$, and the
same lemma gives the parameter inequalities. The partition is used
only for this comparison, so it changes neither $J$ nor the degree
bound obtained from the original factors.
\end{proof}

\begin{theorem}[Total-depth sparsification]
\label{thm:total-depth-sparsification}
Let $H$ have the edge description~\eqref{eq:factor-edge-union},
and let fixed constants $d_0\ge0$, $d_1\ge1$, $\kappa\ge1$ satisfy
conditions~\ref{hyp:retained-degree}--\ref{hyp:biclique-ratio}.
For every fixed $\varepsilon>0$
one can construct deterministically in polynomial time a spanning
subgraph $J\subseteq H$ whose maximum degree depends only on
$d_0,d_1,\kappa,\varepsilon$ and such that
$\totd(H)-\varepsilon N^2\le\totd(J)\le\totd(H)$.
Every elimination forest $F$ of $J$ can be transformed into an
elimination forest $\widehat F$ of $H$ satisfying
$\totd(\widehat F)\le\totd(F)+\varepsilon N^2$.
\end{theorem}

\begin{proof}
Use the same factor replacements as in
Theorem~\ref{thm:td-sparsification}, with
$\zeta=\varepsilon/(4d_1)$. Equation~\eqref{eq:common-forest-exception-budget}
gives $\Delta_{\mathrm{td}}\le\varepsilon N$, and
Lemma~\ref{lem:total-depth-comparison} completes every forest with
total-depth increase at most
$N\Delta_{\mathrm{td}}\le\varepsilon N^2$.
Minimizing therefore gives the lower bound on $\totd(J)$. The upper
bound follows from $J\subseteq H$.
\end{proof}

If a construction has $d$ designated vertices that are not needed
for sparsification, deleting them lowers treedepth and treewidth by
at most $d$. Restoring them above all existing roots increases total
depth by at most $dN$, where $N$ is the original graph order.
These additive terms do not change the linear or quadratic leading
terms in our hardness gaps. We use this observation to omit the three
vertices $z_A,z_B,z_C$ from the NP treedepth construction.
