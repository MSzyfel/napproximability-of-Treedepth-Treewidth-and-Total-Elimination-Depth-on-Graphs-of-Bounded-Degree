The source derivations follow the order in which the paper uses them:

\subsection{The balanced multipartite source}
\label{app:multipartite-normalization}

\begin{restatedtheorem}{Theorem~\ref{thm:ugc-tripartite-cover}}
For every fixed integer $k\geq3$ and fixed
$0<\zeta<1/12$, there is a constant $D_{k,\zeta}$ such that the
following promise problem is Unique Games-hard. The input is a
simple unweighted $k$-partite graph $G$ on $N\geq k$ vertices, of maximum
degree at most $D_{k,\zeta}$,
with a specified partition into $k$ independent parts each of size
$N/k$.
\[
 \textup{YES:}\quad\vc(G)\leq(1/2+\zeta)N,
 \qquad
 \textup{NO:}\quad\vc(G)\geq(1-1/k-\zeta)N.
\]
The reduction is deterministic and polynomial time.
\end{restatedtheorem}

\begin{proof}[Proof of Theorem~\ref{thm:ugc-tripartite-cover}]
Apply the bounded-degree source of Theorem~\ref{thm:bounded-degree-kr-source} with
$\eta=\zeta/2$. This is an admissible parameter,
since $0<\zeta<1/12$ gives $0<\eta<1/24<1/10$.
For a source graph $G$ on $n$ vertices, form $P_k(G)$ as in
Lemma~\ref{lem:multipartite-product}. It has $N=kn$ vertices,
with $n=N/k$ vertices in each of its $k$ independent layers.
The degree identity in the same lemma gives
\[
 \Delta(P_k(G))
 \overset{\text{Lem.~\ref{lem:multipartite-product}}}{=}(k-1)\Delta(G)
 \overset{\text{Thm.~\ref{thm:bounded-degree-kr-source}}}{\leq}
 (k-1)d_{\zeta/2}=:D_{k,\zeta},
\]
which is constant for fixed $k,\zeta$.

The independent-set bounds in Lemma~\ref{lem:multipartite-product} are
\begin{equation}\label{eq:multipartite-product-alpha}
 k\alphaI(G)
 \overset{\text{Lem.~\ref{lem:multipartite-product}}}{\leq}\alphaI(P_k(G))
 \overset{\text{Lem.~\ref{lem:multipartite-product}}}{\leq}
 n+(k-1)\alphaI(G).
\end{equation}
The source bounds are
\begin{equation}\label{eq:multipartite-source-alpha}
 \textup{YES:}\quad
 \alphaI(G)\overset{\text{Thm.~\ref{thm:bounded-degree-kr-source}}}{\geq}
 (1/2-\eta)n,
 \qquad
 \textup{NO:}\quad
 \alphaI(G)\overset{\text{Thm.~\ref{thm:bounded-degree-kr-source}}}{<}\eta n.
\end{equation}
Taking complements and using
$N=kn$ therefore gives both cover thresholds:
\[
\begin{aligned}
 \textup{YES:}\quad\vc(P_k(G))
 &\overset{\eqref{eq:multipartite-product-alpha},\,\eqref{eq:multipartite-source-alpha}}{\leq}
 N-k(1/2-\eta)n
 =(1/2+\eta)N\leq(1/2+\zeta)N,\\
 \textup{NO:}\quad\vc(P_k(G))
 &\overset{\eqref{eq:multipartite-product-alpha},\,\eqref{eq:multipartite-source-alpha}}{>}
 N-n-(k-1)\eta n
  =\left(1-\frac1k-\frac{k-1}{k}\eta\right)N
 \geq\left(1-\frac1k-\zeta\right)N.\end{aligned}
\]
Here $\vc(P_k(G))\overset{\eqref{eq:cover-independent-complement}}{=}N-\alphaI(P_k(G))$ and
$(k-1)\eta/k\leq\eta=\zeta/2<\zeta$, so the strict NO estimate
implies the stated non-strict threshold. The product has no
loops because edges require distinct layer indices, and no parallel
edges because $G$ is simple. Its $k$ is fixed, so its size is
polynomial in that of $G$. Both the independent-set source reduction
and the product construction are deterministic.
Writing $\Lambda$ for the Unique Games input, this normalization is
\[
 \Lambda
 \xrightarrow{\text{Theorem~\ref{thm:bounded-degree-kr-source}, $\eta=\zeta/2$}}G
 \xrightarrow{\text{Lemma~\ref{lem:multipartite-product}}}P_k(G).
\]
\end{proof}

\subsection{The global SSE partition promise}
\label{app:sse-normalization}

\subsubsection{The RST partition theorem}

Use the weighted graph notation of Section~\ref{sec:sseh} and the
Gaussian function $\Gamma_\xi$ defined in
Section~\ref{app:bounded-degree-ugc-source}. We invoke the following
source theorem directly.

\begin{theorem}[RST partition theorem
{\cite[Theorem~3.5 and Section~6.3]{RST}}]
\label{thm:rst-partition}
There is an absolute $\varepsilon_*>0$ such that, for every fixed
integer $q\geq2$, $0<\varepsilon<\varepsilon_*$, and $\gamma>0$,
it is SSE-hard to distinguish the following cases for a finite
undirected weighted graph $G$ with total edge weight $W=w(E(G))>0$.
\begin{description}
\item[YES:] There is a partition
 $V(G)=V_1\mathbin{\dot\cup}\cdots\mathbin{\dot\cup}V_q$ with
 $\mu_G(V_i)=1/q$ and
 $\Phi_G(V_i)\leq\varepsilon+O(\varepsilon^2)$ for every $i\in[q]$.
 The implied constant is absolute.
\item[NO:] Every $A\subseteq V(G)$ with $\mu_G(A)>0$ satisfies
 \[
  \Phi_G(A)\geq
  1-\frac{\Gamma_{1-\varepsilon/2}(\mu_G(A))+\gamma}{\mu_G(A)}.
 \]
\end{description}
\end{theorem}

\begin{lemma}[Gaussian separation at medium measure
{\cite[proof of Corollary~A.4]{RST}}]
\label{lem:gaussian-medium-separation}
There are absolute constants $a,\varepsilon_{\mathrm G}>0$ such that,
for $0<\varepsilon<\varepsilon_{\mathrm G}$ and
$1/10\leq\mu\leq1/2$,
\[
 \mu-\Gamma_{1-\varepsilon/2}(\mu)\geq a\sqrt\varepsilon.
\]
\end{lemma}

\begin{theorem}[Global SSE partition gap, weighted normalized form]
\label{thm:sse-source}
There are absolute constants $c>0$ and $\varepsilon_0>0$ such that,
for every fixed integer $q\geq2$ and every fixed
$0<\varepsilon<\varepsilon_0$, the following promise on finite
undirected graphs $G$ with positive edge weights is SSE-hard.
Parallel edges and loops are allowed, with $W=w(E(G))>0$.
\begin{description}
\item[YES:] There is a partition
  $V(G)=V_1\mathbin{\dot\cup}\cdots\mathbin{\dot\cup}V_q$ with
  $\mu_G(V_i)=1/q$ and $\Phi_G(V_i)\leq2\varepsilon$ for every $i\in[q]$.
\item[NO:] Every $A\subseteq V(G)$ with
  $\mu_G(A)\in[1/10,9/10]$ satisfies
  $w(\cut_G(A))\geq c\sqrt{\varepsilon}\,W$.
\end{description}
\end{theorem}

\begin{proof}
Let $a$ be the constant from Lemma~\ref{lem:gaussian-medium-separation}
and choose $\gamma=a\sqrt\varepsilon/2$ in
Theorem~\ref{thm:rst-partition}.
Choose an absolute $\varepsilon_0\leq\min\{\varepsilon_*,\varepsilon_{\mathrm G}\}$
small enough that its YES bound is at most $2\varepsilon$.
In the NO case, for $\mu=\mu_G(A)\in[1/10,1/2]$,
\[
 \frac{w(\cut_G(A))}{W}
 =2\mu\Phi_G(A)
 \overset{\text{Thm.~\ref{thm:rst-partition}}}{\geq}
 2\bigl(\mu-\Gamma_{1-\varepsilon/2}(\mu)\bigr)-2\gamma
 \overset{\text{Lem.~\ref{lem:gaussian-medium-separation}}}{\geq}
 a\sqrt\varepsilon.
\]
For $\mu_G(A)\in[1/2,9/10]$, apply the same bound to its
complement, which has measure $1-\mu_G(A)\in[1/10,1/2]$ and the
same cut. Thus $c=a$ suffices.
\end{proof}

\subsubsection{Rounding to an unweighted host}

The following conversion proves Lemma~\ref{lem:sse-unweighted-gap}
by replacing weighted edges with parallel unweighted copies.
The source promise uses the wider interval
$[1/10,9/10]$ rather than the interval $[1/4,3/4]$ required by
Lemma~\ref{lem:linegraph-separators} to absorb additive errors in
volumes and cuts. Section~\ref{subsec:bounded-degree-consequences}
later reduces the line graph's degree through incidence clique factors.

\begin{restatedlemma}{Lemma~\ref{lem:sse-unweighted-gap}}
There are absolute constants $c>0$ and $\varepsilon_0>0$ such that,
for every fixed integer $q\geq2$ and every fixed
$0<\varepsilon<\varepsilon_0$, the following promise on unweighted
multigraphs $G$ with $m>0$ edges, allowing loops, is SSE-hard:
\begin{description}
\item[YES:] There is a partition
  $V(G)=V_1\mathbin{\dot\cup}\cdots\mathbin{\dot\cup}V_q$ such that
  $(1-\varepsilon)/q\leq\mu_G(V_i)\leq(1+\varepsilon)/q$ and
  $\Phi_G(V_i)\leq3\varepsilon$ for every $i\in[q]$.
\item[NO:] Every $A\subseteq V(G)$ with
  $\mu_G(A)\in[1/4,3/4]$ satisfies
  $|\cut_G(A)|\geq(c/2)\sqrt{\varepsilon}\,m$.
\end{description}
\end{restatedlemma}

\begin{proof}[Proof of Lemma~\ref{lem:sse-unweighted-gap}]
Let $c$ and $\varepsilon_0$ be the constants from
Theorem~\ref{thm:sse-source}. Decrease $\varepsilon_0$ so that every
$0<\varepsilon<\varepsilon_0$ satisfies $\varepsilon\leq1/10$ and
$\varepsilon^2\leq(c/2)\sqrt\varepsilon$.
Fix $q\geq2$ and $\varepsilon$ in this range, and set
$\theta=\min\{\varepsilon^2,\varepsilon/(2q)\}$.

\emph{Construction.}
Let $G$ be a weighted source instance with $k=|E(G)|\geq1$ and
$W=w(E(G))$, counting parallel edges separately.
Set $M=\lceil k/\theta\rceil$ and
$r_e=\lfloor Mw_e/W\rfloor$ for every $e\in E(G)$.
Form the unweighted multigraph $G^*$ on $V(G)$ by replacing each
edge $e$ by $r_e$ copies with the same endpoints, with no other edges.
A loop is replaced by loops, and an edge with $r_e=0$ is removed.
Write $m=\sum_{e\in E(G)}r_e$ and output $G^*$.

\emph{Rounding estimates.}
For each input edge $e$, write $\omega_e=Mw_e/W-r_e\in[0,1)$.
Since $k\leq\theta M$ and $\omega_e<1$ for each of the $k$ input edges,
\begin{equation}\label{eq:sse-round-edge-count}
 (1-\theta)M\leq M-k<M-\sum_{e\in E(G)}\omega_e=m\leq M.
\end{equation}
Thus $m>0$, and $G^*$ has at most
$M\leq k/\theta+1\leq(1+1/\theta)k$ edges.
The vertex set is unchanged, including vertices whose rounded degree
becomes zero. Since $\theta$ is fixed for each $(q,\varepsilon)$,
the construction is polynomial time and the output has polynomial size.

For a fixed $A\subseteq V(G)=V(G^*)$, an edge contributes zero, one, or two
endpoint incidences to its volume. A loop in $A$ contributes two.
Since $w_e/W-r_e/M=\omega_e/M$, edge $e$ contributes between zero
and $\omega_e/M$ to the difference of normalized volumes. Summing gives
\begin{equation}\label{eq:round-volume}
 0\leq\frac{\vol_G(A)}{2W}
       -\frac{\vol_{G^*}(A)}{2M}
 \leq\frac1M\sum_{e\in E(G)}\omega_e
 <\frac{k}{M}\leq\theta.
\end{equation}

For the cut, only crossing edges contribute. Thus
\begin{equation}\label{eq:round-cut}
  |\cut_{G^*}(A)|
  =M\frac{w(\cut_G(A))}{W}-\sum_{e\in\cut_G(A)}\omega_e
  \geq M\left(\frac{w(\cut_G(A))}{W}-\theta\right),
\end{equation}
because $\sum_{e\in\cut_G(A)}\omega_e<k\leq\theta M$.

\emph{YES case.}
Use the source partition $(V_1,\ldots,V_q)$.
Since $\mu_G(V_i)\overset{\text{Thm.~\ref{thm:sse-source}}}{=}1/q$,
equation~\eqref{eq:round-volume} and
$(1-\theta)M\leq m\leq M$ give
\[
\mu_{G^*}(V_i)
 \overset{\eqref{eq:round-volume},\,\eqref{eq:sse-round-edge-count}}{\geq}
 1/q-\theta
 \geq(1-\varepsilon/2)/q
 \geq(1-\varepsilon)/q,\qquad \mu_{G^*}(V_i)
 \overset{\eqref{eq:round-volume},\,\eqref{eq:sse-round-edge-count}}{\leq}
 \frac{1/q}{1-\theta}
 \leq\frac{1+\varepsilon}{q}.
\]
The lower bound uses $\theta\leq\varepsilon/(2q)$; the upper bound
uses $\theta\leq\varepsilon/(2q)\leq\varepsilon/(1+\varepsilon)$.
The source bounds give
\begin{equation}\label{eq:sse-yes-source-cut}
 w(\cut_G(V_i))=\Phi_G(V_i)\vol_G(V_i)
 \overset{\text{Thm.~\ref{thm:sse-source}}}{\leq}
 2\varepsilon\cdot2W/q=4\varepsilon W/q.
\end{equation}
Since $r_e\leq Mw_e/W$, summing the rounded edge multiplicities over
the cut gives
\begin{equation}\label{eq:sse-yes-rounded-cut}
 |\cut_{G^*}(V_i)|\leq Mw(\cut_G(V_i))/W
 \overset{\eqref{eq:sse-yes-source-cut}}{\leq}4\varepsilon M/q.
\end{equation}
Consequently
\[
\Phi_{G^*}(V_i)
  \overset{\eqref{eq:sse-yes-rounded-cut},\,\eqref{eq:round-volume}}{\leq}
  \frac{4\varepsilon M/q}{2M(1/q-\theta)}
  =\frac{2\varepsilon}{1-q\theta}
 \leq\frac{2\varepsilon}{1-\varepsilon/2}
  <3\varepsilon.
\]
The last inequality uses $\varepsilon\le1/10$.

\emph{NO case.}
Let $A$ be any set with
$\mu_{G^*}(A)\in[1/4,3/4]$. Since $1-\theta\leq m/M\leq1$,
equation~\eqref{eq:round-volume} gives
\[
 \frac{1-\theta}{4}
 \overset{\eqref{eq:sse-round-edge-count}}{\leq}\frac{m}{M}\mu_{G^*}(A)
 \overset{\eqref{eq:round-volume}}{\leq}\mu_G(A)
 \overset{\eqref{eq:round-volume}}{\leq}\frac{m}{M}\mu_{G^*}(A)+\theta
 \overset{\eqref{eq:sse-round-edge-count}}{\leq}\frac34+\theta.
\]
For $\theta\le1/10$, this interval lies in
$[9/40,17/20]\subset[1/10,9/10]$. The source NO promise gives
\begin{equation}\label{eq:sse-source-no-cut}
 \frac{w(\cut_G(A))}{W}
 \overset{\text{Thm.~\ref{thm:sse-source}}}{\geq}c\sqrt{\varepsilon}.
\end{equation}
Applying
\eqref{eq:round-cut}, $\theta\leq(c/2)\sqrt\varepsilon$, and $M\geq m$ gives
\[
  |\cut_{G^*}(A)|
  \overset{\eqref{eq:round-cut},\,\eqref{eq:sse-source-no-cut}}{\geq}
  M(c\sqrt\varepsilon-\theta)
  \geq(c/2)\sqrt\varepsilon\,M
  \overset{\eqref{eq:sse-round-edge-count}}{\geq}(c/2)\sqrt\varepsilon\,m.
\]
Section~\ref{sec:sseh} constructs the simple line graph
$\mathcal L(G^*)$ in $O(M^2)$ time: it has $m\leq M$ vertices,
one per edge occurrence, including parallel edges and loops.
\end{proof}

For reference, let $G_0$ be a basic SSE instance, $H$ the weighted
output supplied by Theorem~\ref{thm:rst-partition}, and $G^*$ its
unweighted host from Lemma~\ref{lem:sse-unweighted-gap}.
The source normalization is
\[
 G_0
 \xrightarrow{\text{Thm.~\ref{thm:rst-partition}}}
 H
 \xrightarrow{\text{Lem.~\ref{lem:sse-unweighted-gap}}}
 G^*.
\]
Theorem~\ref{thm:sse-source} gives the weighted cut bound, and
Equations~\eqref{eq:round-volume} and~\eqref{eq:round-cut} account
for the last arrow's losses.

\subsection{Provenance of the NP-hardness construction}
\label{subsec:np-provenance}

The reduction in Section~\ref{subsec:np-treedepth} passes from a constraint
system to tripartite Vertex Cover and then to Treedepth, following
Bonnet, Neuen, and Soko\l owski~\cite[Sections~3 and~4]{BNS}. Their
truth-block and assignment-witness construction adapts Bonnet's earlier
treewidth reduction~\cite[full version, Section~3.1]{BonnetTreewidth}.
We inherit two opposite truth blocks per variable, complete bipartite
edges between them, and witnesses for satisfying local assignments.
For each variable in its equation, a witness is adjacent to the truth
block assigning the value opposite to the witness's represented value.
Variable-dependent block
sizes also occur in Bonnet's full version~\cite[Section~3.5]{BonnetTreewidth}.

Here the source predicates are three-variable parity equations. Each
has four satisfying assignments. We add a star on their four
witnesses, place its monochromatic witness $000$ or $111$ in a truth
part, and place the other three in the third part. We choose the block
size $5d_x/6$ by analyzing arbitrary partial assignments: uniformly
completing the unassigned variables bounds the excess witnesses of an
equation by $5r_j/6$, where $r_j$ is its number of unassigned variables.
This is the soundness argument proved in Lemma~\ref{lem:xor-cover}.
The corresponding argument in~\cite[proof of Lemma~10]{BNS} uses
variable-by-variable exchanges in a vertex cover.
Pairing the source with a complemented copy then makes all three
parts have size $3m$. These are the modifications to the earlier
encoding used in this proof.

The anchor graph is the one in~\cite[Lemma~6]{BNS}. We use their
identity as a black box at its stated anchor length, together
with the cover-first upper bound and Claims~7 and~9 of their proof.
The main-text corollary extends the identity to anchors as short
as the largest part. Its derivation is given in
Appendix~\ref{app:anchor-proof}. For our balanced source graph this
allows $\ell=3m$ even though
$\vc(G_F)\overset{\text{Lem.~\ref{lem:xor-cover}}}{\geq}11m/2$. Combined with the
parity-witness bound, the anchor identity yields the factor below
$18/17$.

\subsection{The parity source normalization}
\label{app:xor-normalization}

\begin{lemma}[Balancing parity right-hand sides]\label{lem:parity-balance}
A system $F_0$ of $m_0\ge1$ parity equations, each on three distinct
Boolean variables, can be paired
with a system $F_0'$ on fresh variables so that their union has
$2m_0$ equations, optimum $2\OPT(F_0)$, and $m_0$ equations with
each right-hand side. Each variable and its copy retain the
original variable's occurrence count.
\end{lemma}

\begin{proof}
Make a copy $F_0'$ on fresh
variables, replacing every equation $x+y+z=a\pmod2$, with
$a\in\{0,1\}$, by
$x'+y'+z'=1-a$. For an assignment $\sigma$ to $F_0$, assign
$x'=1-\sigma(x)$ in $F_0'$. Since each equation contains three variables,
\[
 (1-\sigma(x))+(1-\sigma(y))+(1-\sigma(z))
 \equiv1-(\sigma(x)+\sigma(y)+\sigma(z))\pmod2.
\]
Thus the copied equation is satisfied if and only if the original
one is satisfied. Complementing all variable values is a bijection
between assignments of the two systems, so
$\OPT(F_0')=\OPT(F_0)$.

Because the variable sets are disjoint, any assignment to the union
satisfies at most $\OPT(F_0)+\OPT(F_0')=2\OPT(F_0)$ equations;
independent optimal assignments attain this bound. The union has
$2m_0$ equations. Every original equation is paired with one having
the opposite right-hand side, so each right-hand side occurs in
$m_0$ equations. Copying to fresh variables preserves each
occurrence count and the distinctness of the three variables in
every equation.
\end{proof}

\begin{lemma}[Repeating parity equations]\label{lem:parity-repetition}
Let $F_1$ be an unweighted system of $m_1\ge1$ equations, and let
$F$ repeat every equation of $F_1$ six times. Then $F$ has
$m=6m_1$ equations and $\OPT(F)=6\OPT(F_1)$. Every occurrence count
is multiplied by six. The fraction of satisfied equations and the
fraction with each right-hand side are unchanged.
\end{lemma}

\begin{proof}
Every assignment satisfies all six copies of an equation or none.
Thus its number of satisfied equations is multiplied by six, as
are the total equation count, every variable's occurrence count,
and the number of equations with either right-hand side.
Taking a maximum over the same assignments proves the optimum claim.
\end{proof}

\begin{restatedtheorem}{Theorem~\ref{thm:xor-source}}
For every fixed $0<\epsilon<1/4$, there is a constant
$b_\epsilon\geq6$ depending only on $\epsilon$ such that the following
promise problem is NP-hard. The input is an unweighted system $F$ of
$m\geq1$ equations $x+y+z=a\pmod2$, where $a\in\{0,1\}$ and
$x,y,z$ are distinct Boolean variables. Equations may be repeated.
Let $\mathcal X(F)$ be its finite Boolean variable set.
Every variable occurs at most $b_\epsilon$ times, with at most two
possible occurrence counts. For $x\in\mathcal X(F)$, write $d_x$ for
the number of equations containing $x$. For $1\leq j\leq m$, write
$a_j$ for the right-hand side of equation $j$,
and let $\OPT(F)$ count the maximum number of satisfied equations,
including repetitions. The instances satisfy
\[
d_x\in6\mathbb Z_{>0},\quad
 \sum_{x\in\mathcal X(F)}d_x=3m,\quad |\{j\in\{1,\ldots,m\}:a_j=0\}|
 =|\{j\in\{1,\ldots,m\}:a_j=1\}|=m/2.
\]
\[
 \textup{YES:}\quad\OPT(F)\geq(1-\epsilon)m,
 \qquad
 \textup{NO:}\quad\OPT(F)\leq(1/2+\epsilon)m.
\]
\end{restatedtheorem}

\begin{proof}[Proof of the normalized form in Theorem~\ref{thm:xor-source}]
Hazan, Safra, and Schwartz state the required source gap as
Theorem~10 of~\cite{HazanSafraSchwartz2003}, attributing it to
H\aa stad. Apply their Max-3-LIN-$q$ theorem with $q=2$ and error
$\epsilon/2$, and call its unweighted output $F_{\mathrm{raw}}$.
There is a fixed integer $B_\epsilon\ge1$ bounding its occurrence
counts; each variable's count is chosen from at most two values,
and no variable repeats within an equation. Thus each equation
has three distinct Boolean variables. The cited YES
threshold $1-\epsilon/2$ is at least $1-\epsilon$, and its NO bound
$<1/2+\epsilon/2$ implies the required bound $\leq1/2+\epsilon$.
We retain this source promise and impose only the additional balance
and divisibility conditions.

Delete unused variables from $F_{\mathrm{raw}}$, calling the result
$F_0$ and its equation count $m_0\ge1$. This changes neither the
equations nor the optimum, and it removes only zero occurrence
counts. Let $F_1=F_0\mathbin{\dot\cup}F_0'$ be the union supplied
by Lemma~\ref{lem:parity-balance}, and let $F$ be the sixfold
repetition of $F_1$ from Lemma~\ref{lem:parity-repetition}.
Record the variable and equation pairing between $F_0$ and its
complemented copy, pairing corresponding repeated equations as well.
This construction data remains available to the later reduction;
it imposes no additional condition on the original source promise.
The normalization is summarized by
\[
 \text{NP-hard gap}
 \xrightarrow{\text{source}}
 F_{\mathrm{raw}}
 \xrightarrow{\text{remove unused}}F_0
 \xrightarrow{\text{balance}}F_1
 \xrightarrow{\text{repeat}}F.
\]

The two lemmas give
\[
m\overset{\text{Lems.~\ref{lem:parity-balance}, \ref{lem:parity-repetition}}}{=}12m_0,
 \OPT(F)\overset{\text{Lems.~\ref{lem:parity-balance}, \ref{lem:parity-repetition}}}{=}12\OPT(F_0),\qquad \frac{\OPT(F)}m=\frac{\OPT(F_0)}{m_0}.
\]
Thus $F$ retains the required YES and NO fractions.

Each remaining variable occurred at least once before repetition,
and its final occurrence count is six times a positive integer.
Set $b_\epsilon=6B_\epsilon\ge6$. Copying preserves occurrence
counts, and repetition multiplies them by six, so there are still
at most two possible positive counts and all are at most
$b_\epsilon$. Thus $d_x\in6\mathbb Z_{>0}$. Counting the three variable
occurrences in every equation gives
$\sum_{x\in\mathcal X(F)}d_x=3m$.
The repetition creates $6m_0=m/2$ equations with each right-hand
side, so the required balance also holds and $m$ is even. These
operations increase the number of equations by the fixed factor
twelve, preserve the distinct variables within each equation, and
are deterministic and polynomial time. This proves all the
conditions in~\eqref{eq:xor-normalization}.
\end{proof}
