\section{Hardness under the Small Set Expansion Hypothesis}
\label{sec:sseh}

We prove that the Small Set Expansion Hypothesis rules out every fixed
approximation factor for treedepth, using the global partition promise
of Raghavendra, Steurer, and Tulsiani (RST)~\cite{RST}. We call the
source graph the \emph{host}. Its line graph has one vertex per host
edge, so both bounds are expressed in terms of the output order: the
YES partition gives a shallow elimination forest, and the NO cut
bound forces a large balanced separator.
Appendix~\ref{app:sse-normalization} derives the unweighted host
promise by rounding the weighted source.

\subsection{The source partition promise}

For a finite undirected weighted graph $G$, let
$w_e>0$ be the weight of edge $e$. Parallel edges
and loops are allowed. For a multiset $F$ of edges, define
\[
 w(F)=\sum_{e\in F}w_e,\qquad W=w(E(G))>0,\qquad
 \deg_G(v)=\sum_{e\in E(G)}m_v(e)w_e,
\]
where edge sums count occurrences, including parallel copies, and,
for an edge $e$ with endpoints $x,y$,
$m_v(e)=\mathbf1_{\{x=v\}}+\mathbf1_{\{y=v\}}$; here
$\mathbf1_{\{P\}}$ is one if $P$ holds and zero otherwise.
For $A\subseteq V(G)$, define $\vol_G(A)=\sum_{v\in A}\deg_G(v)$ and
$\mu_G(A)=\vol_G(A)/(2W)$. When $\vol_G(A)>0$, put
$\Phi_G(A)=w(\cut_G(A))/\vol_G(A)$. A loop contributes
twice its weight to the volume and nothing to a cut.
For an unweighted graph with an edge, take $w_e=1$ for every $e\in E(G)$,
so $W=|E(G)|$.

We use the following precise form of the hypothesis and of hardness
\cite[Hypothesis~5.4 and Definition~5.6]{RST}. For fixed constants
$0<\eta<1/2$ and $0<\delta<1$, the promise problem
$\mathrm{SSE}(\eta,\delta)$ has as input an unweighted regular graph
$G$ of positive degree. Its YES case has a set $A\subseteq V(G)$ with
$\mu_G(A)=\delta$ and $\Phi_G(A)\leq\eta$; its NO case has
$\Phi_G(A)>1-\eta$ for every set of that measure. The
\emph{Small Set Expansion Hypothesis} (SSEH) says that for every such $\eta$
there is a constant $\delta\in(0,\eta)$ for which this promise problem
is NP-hard. A promise problem is \emph{SSE-hard} if, for some fixed
$0<\eta<1/2$ and every fixed $\delta\in(0,\eta)$, there is a deterministic
polynomial-time reduction from $\mathrm{SSE}(\eta,\delta)$ that maps
its YES and NO cases to the respective cases of the target promise.
The polynomial may depend on these fixed parameters. Consequently,
an SSE-hard promise problem is NP-hard under SSEH.

\begin{lemma}[Global SSE partition gap, normalized form]\label{lem:sse-unweighted-gap}
There are absolute constants $c>0$ and $\varepsilon_0>0$ such that,
for every fixed integer $q\geq2$ and every fixed
$0<\varepsilon<\varepsilon_0$, the following promise on unweighted
multigraphs $G$ with $m>0$ edges, allowing loops, is SSE-hard:
\begin{description}
\item[YES:] There is a partition
  $V(G)=V_1\mathbin{\dot\cup}\cdots\mathbin{\dot\cup}V_q$ such that
  $(1-\varepsilon)/q\leq\mu_G(V_i)\leq(1+\varepsilon)/q$ and
  $\Phi_G(V_i)\leq3\varepsilon$ for every $i\in[q]$.
\item[NO:] Every $A\subseteq V(G)$ with
  $\mu_G(A)\in[1/4,3/4]$ satisfies
  $|\cut_G(A)|\geq(c/2)\sqrt{\varepsilon}\,m$.
\end{description}
\end{lemma}

The host degree is unrestricted. Section~\ref{subsec:bounded-degree-consequences}
later reduces the line graph's degree through incidence clique
factors. The following lemma counts crossing and internal edge weights.

\begin{lemma}[Crossing and internal edge weights]
\label{lem:sse-yes-edge-counts}
Let $G$ be a finite undirected weighted multigraph with positive
edge weights and total edge weight $W>0$.
Let $(V_1,\ldots,V_q)$, where $q\geq1$ is an integer, be a partition
of its vertices into parts of positive volume.
Let $D$ be the multiset of edges whose endpoints lie in different
classes. For each $i\in[q]$, let $E_i$ be the multiset of edges with both
endpoints in $V_i$, including loops at vertices of $V_i$. Each
parallel edge is included in the corresponding multiset. Then
$w(D)\leq W\max_{i\in[q]}\Phi_G(V_i)$ and
$w(E_i)\leq\vol_G(V_i)/2$ for every $i\in[q]$.
\end{lemma}

\begin{proof}
Each edge in $D$ belongs to the cuts of its two endpoint classes.
Internal edges, including loops, belong to none. Hence
\[
  2w(D)=\sum_{i=1}^q w(\cut_G(V_i))
  =\sum_{i=1}^q\Phi_G(V_i)\vol_G(V_i)
  \leq\max_{i\in[q]}\Phi_G(V_i)\sum_{i=1}^q\vol_G(V_i)
  =2W\max_{i\in[q]}\Phi_G(V_i).
\]
Each internal edge, including a loop, contributes twice its weight
to its class volume, whereas a crossing edge contributes once. Thus
$2w(E_i)+w(\cut_G(V_i))=\vol_G(V_i)$, which gives the second bound.
\end{proof}

\subsection{The line graph and its separator lower bound}

For an unweighted host multigraph $G$, put $m=|E(G)|$ and define
its \emph{line graph} by
\[
 V(\mathcal L(G))=E(G),\qquad
 E(\mathcal L(G))=\{\{e,f\}:e,f\in E(G),\ e\ne f,\ e\cap f\ne\varnothing\},
\]
where $E(G)$ contains edge occurrences and $e\cap f\ne\varnothing$
means that the two occurrences share an endpoint. Parallel host
edges therefore give distinct vertices.
This is a simple unweighted graph, even when the host has
parallel edges or loops. In particular,
$|V(\mathcal L(G))|=|E(G)|=m$. Testing each
pair of distinct host-edge occurrences for a shared endpoint constructs
this graph in $O(m^2)$ time and produces at most $\binom m2$ edges.

\begin{lemma}[A treedepth upper bound from a partition]\label{lem:sse-complete}
Let $G$ be an unweighted host multigraph with $m>0$ edges, counting
parallel edges separately, and let $(V_1,\ldots,V_q)$ be a partition
of its vertices. If $D$ is the multiset of edges joining different
classes, then $\td(\mathcal{L}(G))\leq |D|+\max_{i\in[q]} |E(G[V_i])|$.
\end{lemma}

\begin{proof}
After deleting the line-graph vertices corresponding to $D$, the
remaining vertices partition into the sets $E(G[V_i])$, $i\in[q]$.
Different sets are nonadjacent, because their host edges have endpoints
in disjoint classes. A chain on each nonempty set therefore gives an
elimination forest of height at most $\max_{i\in[q]}|E(G[V_i])|$.
Restoring $D$ above its roots increases the height by $|D|$, by
Lemma~\ref{lem:restore-vertices}. That lemma also covers empty sets
and $D=\varnothing$.
\end{proof}

Deleting vertices of $\mathcal L(G)$ deletes the corresponding host
edge copies. The remaining components have disjoint sets of incident
host vertices, yielding a host cut covered by the deleted edges.

For $0<\rho<1$, a \emph{$\rho$-balanced separator} of an $N$-vertex
graph $H$ is a set $S\subseteq V(H)$ such that every component of
$H-S$ has at most $\rho N$ vertices. We write $b_\rho(H)$ for its
minimum size. A \emph{balanced separator} is a $1/2$-balanced
separator, and we abbreviate $b_{1/2}(H)$ to $b(H)$.

\begin{lemma}[Balanced separators from tree decompositions
{\cite[Lemma~5.3]{APW}}]
\label{lem:treewidth-balanced-separator}
Every finite nonempty graph $H$ satisfies
$b(H)\leq\operatorname{tw}(H)+1$.
\end{lemma}

\begin{lemma}[From host cuts to line-graph separators]
\label{lem:linegraph-separators}
Let $G$ be an unweighted host multigraph with $m>0$ edges, counting
parallel edges separately.
Suppose that, for some $\beta>0$, every set $A\subseteq V(G)$ with
$\mu_G(A)\in[1/4,3/4]$ satisfies
\begin{equation}\label{eq:sse-host-cut-premise}
 |\cut_G(A)|\geq\beta m.
\end{equation}
Then $b_\rho(\mathcal{L}(G))\geq\min\{(3/4-\rho)m,\beta m\}$
for every $1/2\leq\rho<3/4$.
\end{lemma}

\begin{proof}
Fix a $\rho$-balanced separator $S$ of $\mathcal L(G)$, and write
$F_S$ for the corresponding multiset of host edges, so $|F_S|=|S|$.
If $|S|\ge(3/4-\rho)m$, the claim is immediate. Assume henceforth
that $|S|<(3/4-\rho)m$.

\emph{Grouping the surviving host edges.}
The line graph $\mathcal L(G)-S$ has
$m-|S|>(1/4+\rho)m$ vertices, with each component of order at
most $\rho m$. Choose a union $Z$ of components with
$m/4\le|Z|\le\rho m$. If a component has size in this interval,
take it. Otherwise, every component has fewer than $m/4$ vertices.
Add components until their total first reaches $m/4$.
The total is then below $m/2\le\rho m$.

Let $X$ be the host edges corresponding to vertices in $Z$, and
let $Y$ be the other surviving host edges. Then
$|X|=|Z|\ge m/4$ and
$|Y|=m-|S|-|X|>m/4$.
If a host vertex were incident to edges in both multisets, their
line-graph vertices would be adjacent, with one in $Z$ and one
outside $Z$. This contradicts the choice of $Z$ as a union of
components.

\emph{Forming the host cut.}
Let $A,B\subseteq V(G)$ be the sets of endpoints of edges in
$X,Y$, respectively. The preceding paragraph gives
$A\cap B=\varnothing$. Every edge of $X$
contributes two degree units to $\vol_G(A)$, including a loop.
The same holds for $Y$ and $B$. Thus
$\vol_G(A)\ge2|X|\ge m/2$ and
$\vol_G(B)\ge2|Y|\ge m/2$.
Since the total host volume is $2m$,
$\vol_G(A)\le2m-\vol_G(B)\le3m/2$.
Since $\mu_G(A)=\vol_G(A)/(2m)$, we obtain
$\mu_G(A)\in[1/4,3/4]$.
Every surviving edge lies wholly in $A$ or wholly in
$B\subseteq V(G)\setminus A$, so every edge crossing $A$ belongs
to $F_S$. The cut assumption therefore yields
$|S|=|F_S|\ge|\cut_G(A)|
\overset{\eqref{eq:sse-host-cut-premise}}{\ge}\beta m$, proving the bound.
\end{proof}

\begin{lemma}\label{lem:balanced-by-td}
For every finite graph $H$, $b(H)\leq\td(H)$.
Moreover, every elimination forest $F$ of an $N$-vertex graph $H$
satisfies $\totd(F)\geq(N/2)b(H)$.
\end{lemma}

\begin{proof}
Fix an elimination forest $F$ of $H$, and put $N=|V(H)|$.
If every root subtree has
at most $N/2$ vertices, then $b(H)=0$ and both claims follow.
Otherwise, delete the root of the unique subtree of size greater
than $N/2$, and continue into a child subtree of size greater than
$N/2$ while one exists. Such a child is unique, since two disjoint
subtrees of that size cannot fit among $N$ vertices.

The deleted vertices form an ancestor path $P$. Every remaining
subtree has at most $N/2$ vertices, and no graph edge joins distinct
remaining subtrees. Thus $P$ is balanced and
$b(H)\leq|P|\leq h(F)$. Minimizing height proves the first claim.
The original subtree rooted at the last vertex of $P$ has more than
$N/2$ vertices, each at depth at least $|P|\geq b(H)$. Their depths
alone give $\totd(F)\geq(N/2)b(H)$.
\end{proof}

\begin{corollary}[NO treedepth bound]\label{lem:sse-sound}
Under the cut assumption of Lemma~\ref{lem:linegraph-separators},
$\td(\mathcal L(G))\geq\min\{m/4,\beta m\}$.
\end{corollary}

\begin{proof}
Take $\rho=1/2$. Then
\[
 \td(\mathcal L(G))
 \overset{\text{Lem.~\ref{lem:balanced-by-td}}}{\geq}
 b(\mathcal L(G))
 \overset{\text{Lem.~\ref{lem:linegraph-separators}}}{\geq}
 \min\{m/4,\beta m\}.
\]
\end{proof}

Let $G_{\mathrm w}$ be a weighted partition instance from
Theorem~\ref{thm:sse-source},
$G^*$ its rounded unweighted host from
Lemma~\ref{lem:sse-unweighted-gap}, and $H=\mathcal L(G^*)$ its
simple line graph. The construction is
\[
 G_{\mathrm w}\xrightarrow{\text{round weights}}G^*
 \xrightarrow{\text{line graph}}H.
\]
Lemmas~\ref{lem:sse-complete} and~\ref{lem:linegraph-separators}
give its completeness and soundness bounds. The later reduction
in Section~\ref{subsec:bounded-degree-consequences} bounds the degree
of $H$.

\subsection{The approximation gap}
\label{subsec:sse-dense-gap}

\begin{proof}[Dense treedepth gap: proof sketch]
For a small fixed source error $\varepsilon>0$, take
$q=\lceil2/\varepsilon\rceil$ in Lemma~\ref{lem:sse-unweighted-gap}
and let $m$ be the number of host edges. The completeness forest
has height at most $4\varepsilon m$ by
Lemma~\ref{lem:sse-complete}, while
Corollary~\ref{lem:sse-sound} gives NO height at least
$(c/2)\sqrt\varepsilon\,m$ for sufficiently small $\varepsilon$.
The ratio $c/(8\sqrt\varepsilon)$ is unbounded, excluding every
fixed approximation factor. Appendix~\ref{app:dense-sse-td}
derives these bounds and gives explicit parameter conditions and
the decision threshold.
\end{proof}
