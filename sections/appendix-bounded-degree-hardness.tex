\subsection{Derivation of the bounded-degree hardness corollaries}
\label{app:bounded-degree-hardness}

For each promise-gap transfer, fix a target factor $r$ and a larger
dense hardness factor $r'$ in the corresponding range. We use
Lemma~\ref{lem:gap-preservation} after fixing the dense source
parameters. Taking disjoint copies of a parity system on disjoint
variable sets multiplies both its equation count and its optimum
by the number of copies. For independent-set sources, disjoint
copies multiply the order and independence number by the same
integer and preserve maximum degree. An independent set in the
union has at least its average number of selected vertices in one
copy, so a relative approximation also transfers to the original
graph. In the SSE host, replace every
edge occurrence by the same number of parallel copies. This
multiplies all volumes, cut sizes, and the line-graph order by that
number. These operations preserve the source promises and allow
us to require sufficiently large output orders.
In all SSE applications, take $q=\lceil2/\eta\rceil$ as in the
proof of Theorem~\ref{thm:sse-hardness}.

\subsubsection{Treedepth}
\label{app:bounded-degree-treedepth}

For every variable $x$ of $F$ and $a\in\{0,1\}$, let
$W_x^a\subseteq V(G_F)$ consist of the witnesses for equations
containing $x$ whose represented assignments set $x=a$.

\begin{lemma}[Local factors of the parity graph]\label{lem:parity-local-factors}
The edges of $G_F$ are the union of its witness-star edges and the
bicliques $T_x^0\times T_x^1$, $T_x^0\times W_x^1$, and
$T_x^1\times W_x^0$ over all variables $x$.
The star edges have maximum degree three, every vertex belongs to
at most three local factors, and their side-size ratios are at
most $12/5$. Moreover, $W_x^0\subseteq A\cup C$ and
$W_x^1\subseteq B\cup C$ for every variable $x$.
\end{lemma}

\begin{proof}
These are the truth-block and inconsistency edges in the
construction. A truth vertex belongs to two local bicliques. A
witness belongs to one per variable of its equation, hence three.
The sizes $|T_x^a|=5d_x/6$ and $|W_x^a|=2d_x$ give the ratio
$12/5$. A witness setting $x=0$ lies in $A\cup C$, and one
setting $x=1$ lies in $B\cup C$, by the placement of the special
witnesses.
\end{proof}

\begin{lemma}[Clique factors of a line graph]\label{lem:linegraph-factors}
Let $G$ be an unweighted host multigraph. Identify each edge
occurrence $e\in E(G)$ with its vertex in $\mathcal L(G)$, and put
$U_v=\{e\in E(G):v\text{ is an endpoint of }e\}$ for $v\in V(G)$.
The nonempty sets
$U_v$ give clique factors for $\mathcal L(G)$, and each line-graph
vertex belongs to at most two factors.
\end{lemma}

\begin{proof}
Two line-graph vertices are adjacent if and only if their host edges
share an endpoint, so
$E(\mathcal L(G))=\bigcup_v\binom{U_v}{2}$.
An edge has at most two endpoints. A loop contributes one vertex
to the factor at its endpoint.
\end{proof}

\begin{proof}[Proof of Corollary~\ref{cor:td-bounded-degree}]
Apply Theorem~\ref{thm:td-sparsification} with error parameter
$\tau>0$. It preserves the YES upper bound and changes treedepth
by at most $\tau N$ on a graph of order $N$.

\emph{NP-hardness.}
Use the local factors of Lemma~\ref{lem:parity-local-factors},
the three anchor cliques, and, after omitting $z_A,z_B,z_C$, the
three anchor--part bicliques. Set $E_0$ equal to the witness-star
edges. Each vertex of $G_F$ belongs to at most three local factors
and one anchor--part biclique. Each anchor vertex belongs to its
clique and one anchor--part biclique. The anchor bicliques have
equal side sizes, so $d_1=4$, $d_0=3$, and $\kappa=12/5$.

The original dense graph $H_{3m}(G_F)$ contains an anchor clique
together with its vertex $z_X$, giving the lower bound $3m+1$
on its treedepth. Omitting the three vertices and sparsifying the
remaining $18m$ vertices loses at most $3+18\tau m$. This is an
arbitrarily small fraction of $3m+1$ when $\tau$ is small and $m$
is large. Lemma~\ref{lem:gap-preservation} therefore transfers
every factor below $18/17$ from Theorem~\ref{thm:np-hardness}.

\emph{UGC hardness.}
Use Theorem~\ref{thm:bounded-degree-kr-source} in the construction
$S_q(G)$. Replace its $q$ layer cliques and let $E_0$ be its
cross-layer edges. Each vertex belongs to one factor and has degree
at most $(q-1)d_\eta$ in the retained graph. Thus $d_1=1$ and
$d_0=(q-1)d_\eta$, with no biclique factors.
A layer clique gives $\td(S_q(G))\ge n$, while the loss is at most
$\tau qn$. Since $q$ is fixed, choosing $\tau$ sufficiently small
allows Lemma~\ref{lem:gap-preservation} to transfer every factor
below $4/3$ from Theorem~\ref{thm:ugc-hardness}.

\emph{SSE hardness.}
Use the host $G^*$ from Lemma~\ref{lem:sse-unweighted-gap}, with
source parameter $\eta$ and $N$ edges counted with multiplicity.
Lemma~\ref{lem:linegraph-factors} gives the factor description of
$\mathcal L(G^*)$ with $d_1=2$, $d_0=0$, and no biclique factors.
The YES bound is $4\eta N$. Put $\beta=(c/2)\sqrt\eta\le1/4$.
Apply Lemma~\ref{lem:linegraph-separators} with $\rho=1/2$, then
Lemma~\ref{lem:balanced-by-td}, to obtain NO bound $\beta N$.
Since $\beta/(4\eta)$ grows without
bound as $\eta$ decreases, first fix $\eta$ to obtain a dense gap
larger than the target factor. Then choose $\tau$ so that
$\tau N$ is a sufficiently small fraction of $\beta N$.
Lemma~\ref{lem:gap-preservation} preserves the target gap.
\end{proof}

\subsubsection{Treewidth}
\label{app:bounded-degree-treewidth}

\begin{lemma}[Balanced separator from treewidth]
\label{lem:balanced-by-tw}
For every finite nonempty graph $H$,
$b(H)\leq\operatorname{tw}(H)+1$~\cite[Lemma~5.3]{APW}.
\end{lemma}

\begin{proof}[Proof of Corollary~\ref{cor:tw-bounded-degree}]
Apply Theorem~\ref{thm:tw-sparsification} with error parameter
$\tau>0$. It preserves the YES upper bound and changes treewidth
by at most $\tau N$ on a graph of order $N$.

\emph{NP-hardness.}
Use the three global cliques of $K_R(G_F)$ and the local factors
of Lemma~\ref{lem:parity-local-factors} in every source copy.
Let $E_0$ consist of the witness-star edges in these copies.
Thus $d_1=4$, $d_0=3$, and $\kappa=12/5$.
Each truth block lies in one global clique, while each $W_x^a$
meets at most two. Partitioning these sets by the global parts
therefore verifies the clique-containment hypothesis with $s=2$.
The graph has order $9Rm$ and contains a clique of order $3Rm$,
so its treewidth is at least $3Rm-1$. For fixed $R$, the loss
$9\tau Rm$ is an arbitrarily small fraction of this lower bound.
Lemma~\ref{lem:gap-preservation} transfers every factor below
$12/11$ from Theorem~\ref{thm:tw-hardness}.

\emph{UGC hardness.}
For the source of Theorem~\ref{thm:bounded-degree-kr-source}, put
$Q=P_3(G)$ as in Lemma~\ref{lem:multipartite-product}. In $K_R(Q)$,
replace the three global cliques and retain the edges within the
source copies. Each vertex belongs to one factor, and its retained
degree is at most $2d_\eta$. Thus $d_1=1$, $d_0=2d_\eta$, and
there are no biclique factors. The product lemma gives
$\vc(Q)\le(3/2+3\eta)n$ in the YES case and
$\vc(Q)\ge(2-2\eta)n$ in the NO case, giving the dense gap of
Theorem~\ref{thm:tw-ugc}.
The graph has order $3Rn$ and a global clique of order $Rn$.
Hence its treewidth is at least $Rn-1$, while the loss is at most
$3\tau Rn$. With the source parameters fixed, choose $\tau$ small
and $n$ large and apply Lemma~\ref{lem:gap-preservation} to preserve
every target factor below $4/3$.

\emph{SSE hardness.}
Use the line graph of the host in Lemma~\ref{lem:sse-unweighted-gap},
with $N$ host edges. Lemma~\ref{lem:linegraph-factors} gives
$d_1=2$, $d_0=0$, and no biclique factors. The dense YES width is
at most its height bound $4\eta N$. With
$\beta=(c/2)\sqrt\eta\le1/4$, apply
Lemma~\ref{lem:linegraph-separators} with $\rho=1/2$ and then
Lemma~\ref{lem:balanced-by-tw} to obtain NO bound $\beta N-1$.
First choose $\eta$ to make $\beta/(4\eta)$ larger than the target
factor. The sparse NO bound is at least $\beta N-(1+\tau N)$.
For small $\tau$ and large $N$, the term $1+\tau N$ is an
arbitrarily small fraction of $\beta N$. Thus
Lemma~\ref{lem:gap-preservation} applies.
\end{proof}

\subsubsection{Total depth}
\label{app:bounded-degree-total-depth}

\begin{proof}[Proof of Corollary~\ref{cor:total-depth-bounded-degree}]
Apply Theorem~\ref{thm:total-depth-sparsification} with error
parameter $\tau>0$. It changes total depth by at most $\tau N^2$
on a graph of order $N$, and a sparse forest can be completed at
the same additive cost.

\emph{NP-hardness.}
Let $\delta_0>0$ be the constant in
Theorem~\ref{thm:source-cubic-independent-set}.
Choose a fixed rational
$0<\epsilon_0<\min\{1/200,\delta_0/184\}$ and an integer
$\Lambda\ge64/\epsilon_0$. For a cubic source $J$ on $n\ge4$ vertices,
put $H=H^{\Sigma}_{\Lambda n}(G(J))$. The six subgraphs used in sparsification
are the three cliques on the parts of $G(J)$ and the three complete
bipartite graphs joining each part $X\in\{A,B,C\}$ to $T_X$. The
parts have sizes $2n,9n/2,9n/2$, and
$|T_A|=|T_B|=|T_C|=\Lambda n$. A vertex
of $G(J)$ belongs to its part clique and the corresponding complete
bipartite graph. Each added vertex belongs only to that graph. Since
$\Lambda\ge5$,
the largest biclique side-size ratio is $\Lambda/2$.
Set $E_0=E(G(J))$, so $(V(H),E_0)$ has maximum degree at most three.
Thus the theorem applies
with $d_1=2$, $\kappa=\Lambda/2$, and $d_0=3$.

Write $N_G=|V(G(J))|=11n$ for the base graph order and
$N_H=|V(H)|=N_G+3\Lambda n=(11+3\Lambda)n$ for the output graph order. The size condition of
Lemma~\ref{lem:np-total-depth-extraction} follows from
\[
 \frac{N_G(N_G+1)}{2\Lambda n}=\frac{11n(11n+1)}{2\Lambda n}
 =\frac{121n+11}{2\Lambda}
 \leq\frac{495}{8\Lambda}n<\epsilon_0n.
\]
Here $n\ge4$ gives $11\le(11/4)n$, and
$\Lambda\ge64/\epsilon_0$ gives
$495/(8\Lambda)\le495\epsilon_0/512<\epsilon_0$.
Also, Lemma~\ref{lem:np-total-depth-cubic} and $\alphaI(J)\le n$ give
$W(G(J))=39n/2-\alphaI(J)+3\ge37n/2+3\ge n$. Hence
$\totd(H)\ge\Lambda nW(G(J))\ge\Lambda n^2$, and
$\Lambda n^2=\Lambda N_H^2/(11+3\Lambda)^2$ by the definition of $N_H$.
Thus the optimum is bounded below by a fixed positive multiple of
$N_H^2$. Making $\tau$ small enough therefore makes the additive
completion error at most an $\epsilon_0$ fraction of the optimum.
Choose $\tau\le\epsilon_0\Lambda/(11+3\Lambda)^2$. Then
$\tau N_H^2\le\epsilon_0\Lambda n^2\le\epsilon_0\totd(H)$.
Let $H'\subseteq H$ be the sparse graph. The same bounds give
\[
 \totd(H')\ge\totd(H)-\tau N_H^2
 \ge\Lambda n^2-\epsilon_0\Lambda n^2
 =(1-\epsilon_0)\Lambda n^2.
\]
Also $\totd(H')\le\totd(H)$. Thus the solution part of
Lemma~\ref{lem:gap-preservation}, with $a=1$ and
$L(H)=\Lambda n^2$, converts a $(1+\epsilon_0)$-approximate forest
of $H'$ into a $(1+2\epsilon_0)$-approximate forest of $H$.
Lemma~\ref{lem:np-total-depth-extraction} then yields an independent
set of $J$ of size at least
$(1-184\epsilon_0)\alphaI(J)>(1-\delta_0)\alphaI(J)$, contradicting
the source hardness.

\emph{UGC hardness.}
Use Theorem~\ref{thm:bounded-degree-kr-source} in $S_q(G)$.
Replace its layer cliques and retain its cross-layer edges.
As in the treedepth proof, $d_1=1$, $d_0=(q-1)d_\eta$, and no
biclique factors are needed. Restriction to each of the $q$
disjoint layer cliques gives
\[
 \totd(S_q(G))\ge q\,\frac{n(n+1)}2
 \ge\frac{qn^2}{2}=\frac{N^2}{2q},\qquad N=qn.
\]
The loss $\tau N^2$ is an arbitrarily small fraction of this lower
bound for fixed $q$. Lemma~\ref{lem:gap-preservation} therefore
transfers every factor below $8/7$ from
Theorem~\ref{thm:total-depth-ugc}.

\emph{SSE hardness.}
Use $\mathcal L(G^*)$, with the factors of
Lemma~\ref{lem:linegraph-factors}: $d_1=2$, $d_0=0$, and no
biclique factors. Its order is $N=|E(G^*)|$, counting edge
multiplicity. The YES height bound $4\eta N$ gives total-depth
bound $4\eta N^2$. With $\beta=(c/2)\sqrt\eta\le1/4$,
apply Lemma~\ref{lem:linegraph-separators} with $\rho=1/2$ and then
Lemma~\ref{lem:balanced-by-td} to obtain NO total-depth bound
$\beta N^2/2$.
The ratio $\beta/(8\eta)$ grows without bound as $\eta$ decreases.
First choose $\eta$ for a dense gap larger than the target factor,
then make the loss $\tau N^2$ a sufficiently small fraction of
$\beta N^2/2$. Lemma~\ref{lem:gap-preservation} preserves the gap.
\end{proof}
